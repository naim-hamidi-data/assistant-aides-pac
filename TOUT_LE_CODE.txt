function cloner(objet) {
    return JSON.parse(JSON.stringify(objet));
}

function fusionnerProfondement(cible, source) {
    if (!source || typeof source !== "object") return cible;

    Object.keys(source).forEach(cle => {
        const valeur = source[cle];
        if (
            valeur &&
            typeof valeur === "object" &&
            !Array.isArray(valeur) &&
            cible[cle] &&
            typeof cible[cle] === "object"
        ) {
            fusionnerProfondement(cible[cle], valeur);
        } else {
            cible[cle] = valeur;
        }
    });

    return cible;
}

let profilExploitation = cloner(PROFIL_INITIAL);

function lireProfil(chemin) {
    return chemin.split(".").reduce((objet, cle) => {
        if (objet === null || objet === undefined) return undefined;
        return objet[cle];
    }, profilExploitation);
}

function modifierProfil(chemin, valeur) {
    const parties = chemin.split(".");
    let objet = profilExploitation;

    parties.slice(0, -1).forEach(cle => {
        if (!objet[cle] || typeof objet[cle] !== "object") objet[cle] = {};
        objet = objet[cle];
    });

    objet[parties[parties.length - 1]] = valeur;
}

function sauvegarderProfilLocalement() {
    try {
        localStorage.setItem(CLE_STOCKAGE_PROFIL, JSON.stringify(profilExploitation));
    } catch (erreur) {
        console.warn("Impossible de sauvegarder le profil localement.", erreur);
    }
}

function chargerProfilLocal() {
    try {
        const brut = localStorage.getItem(CLE_STOCKAGE_PROFIL);
        if (!brut) return profilExploitation;
        const sauvegarde = JSON.parse(brut);
        profilExploitation = fusionnerProfondement(cloner(PROFIL_INITIAL), sauvegarde);
    } catch (erreur) {
        console.warn("Impossible de charger le profil local.", erreur);
        profilExploitation = cloner(PROFIL_INITIAL);
    }
    return profilExploitation;
}

function reinitialiserProfil() {
    profilExploitation = cloner(PROFIL_INITIAL);
    try {
        localStorage.removeItem(CLE_STOCKAGE_PROFIL);
    } catch (_) {}
    return profilExploitation;
}

const STATUT_AIDE = {
    A_EXAMINER: "a_examiner",
    INFORMATIONS_MANQUANTES: "informations_manquantes",
    NON_RETENUE: "non_retenue"
};

function creerResultatAide(id, statut, raisons = [], informationsManquantes = [], verifications = [], details = {}) {
    const aide = obtenirAide(id);
    return {
        id,
        nom: aide ? aide.nom : id,
        categorie: aide ? aide.categorie : "",
        statut,
        raisons,
        informationsManquantes,
        verifications,
        details,
        source: aide ? aide.source : ""
    };
}

function nonRetenue(id, raison) {
    return creerResultatAide(id, STATUT_AIDE.NON_RETENUE, [], [], raison ? [raison] : []);
}

function verifierAgriculteurActif() {
    const statut = profilExploitation.agriculteurActif.statut;
    if (statut === true) return { valide: true, inconnu: false };
    if (statut === false) return { valide: false, inconnu: false };
    return { valide: false, inconnu: true };
}

function verifierActifPourAide(id) {
    const actif = verifierAgriculteurActif();
    if (actif.inconnu) {
        return creerResultatAide(
            id,
            STATUT_AIDE.INFORMATIONS_MANQUANTES,
            [],
            ["Statut d'agriculteur actif"]
        );
    }
    if (!actif.valide) return nonRetenue(id, "Le critère d'agriculteur actif n'est pas rempli.");
    return null;
}

function valeurPositive(valeur) {
    return nombreOuZero(valeur) > 0;
}

function niveauScore(niveau) {
    return { base: 1, superieur: 2, bio: 3 }[niveau] || 0;
}

    if (profilExploitation.dpb.detientDPB === null) manquants.push("Détention de DPB");
    if (!estRenseigne(profilExploitation.dpb.nombreDPBActives)) manquants.push("Nombre de DPB activés");

    if (manquants.length) {
        return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], manquants);
    }

    if (!aDesDPBActives() || nombreOuZero(profilExploitation.surfaces.surfaceAdmissible) <= 0) {
        return nonRetenue(id, "Aucun DPB ne peut être activé sur une surface admissible positive.");
    }

    return creerResultatAide(
        id,
        STATUT_AIDE.A_EXAMINER,
        ["Des DPB activés et une surface admissible sont renseignés."],
        [],
        ["La valeur réelle des DPB de l'exploitation reste déterminante pour le montant."]
    );
}

function verifierQualificationJA() {
    const i = profilExploitation.installation;
    const exp = nombreOuZero(i.experienceAgricoleMois);

    if (i.diplomeAgricoleNiveau4OuPlus === true) return { valide: true, inconnu: false };
    if (exp >= 40) return { valide: true, inconnu: false };
    if (i.diplomeNiveau3OuPlus === true && exp >= 24) return { valide: true, inconnu: false };

    const informationsSuffisantes =
        i.diplomeAgricoleNiveau4OuPlus !== null &&
        i.diplomeNiveau3OuPlus !== null &&
        estRenseigne(i.experienceAgricoleMois);

    return { valide: false, inconnu: !informationsSuffisantes };
}

function analyserACJA() {
    const id = "acja";
    const actif = verifierActifPourAide(id);
    if (actif) return actif;

    const i = profilExploitation.installation;
    const manquants = [];

    if (i.premiereInstallation === false) return nonRetenue(id, "La demande ne concerne pas une première installation récente.");
    if (i.premiereInstallation === null) manquants.push("Première installation");
    if (!estRenseigne(i.age)) manquants.push("Âge");
    if (!estRenseigne(i.anneeInstallation)) manquants.push("Année de première installation");

function categorieSignificative(surfaceCategorie, surfaceAdmissible) {
    const total = nombreOuZero(surfaceAdmissible);
    if (total <= 0) return false;
    return nombreOuZero(surfaceCategorie) / total >= 0.05;
}

function evaluerVoiePratiques() {
    const s = profilExploitation.surfaces;
    const e = profilExploitation.ecoregime;
    const manquants = [];
    const niveaux = [];
    let auMoinsUneCategorie = false;

    if (!valeurPositive(s.surfaceAdmissible)) return { eligible: false, manquants: ["Surface admissible"] };

    if (categorieSignificative(s.terresArables, s.surfaceAdmissible)) {
        auMoinsUneCategorie = true;
        if (!estRenseigne(e.pointsDiversification)) manquants.push("Points de diversification des cultures");
        else if (nombreOuZero(e.pointsDiversification) < 4) return { eligible: false, manquants: [], motif: "La diversification des cultures n'atteint pas 4 points." };
        else niveaux.push(nombreOuZero(e.pointsDiversification) >= 5 ? "superieur" : "base");
    }

    if (categorieSignificative(s.prairiesPermanentes, s.surfaceAdmissible)) {
        auMoinsUneCategorie = true;
        if (!estRenseigne(e.pourcentagePrairiesNonLabourees)) manquants.push("Pourcentage de prairies permanentes non labourées");
        else if (nombreOuZero(e.pourcentagePrairiesNonLabourees) < 80) return { eligible: false, manquants: [], motif: "Le taux de prairies permanentes non labourées est inférieur à 80 %." };
        else niveaux.push(nombreOuZero(e.pourcentagePrairiesNonLabourees) >= 90 ? "superieur" : "base");
    }

    if (categorieSignificative(s.culturesPermanentes, s.surfaceAdmissible)) {
        auMoinsUneCategorie = true;
        if (!estRenseigne(e.pourcentageInterRangsCouverts)) manquants.push("Pourcentage d'inter-rangs couverts");
        else if (nombreOuZero(e.pourcentageInterRangsCouverts) < 75) return { eligible: false, manquants: [], motif: "La couverture des inter-rangs est inférieure à 75 %." };
        else niveaux.push(nombreOuZero(e.pourcentageInterRangsCouverts) >= 95 ? "superieur" : "base");
    }

    if (manquants.length) return { eligible: false, manquants };
    if (!auMoinsUneCategorie) return { eligible: false, manquants: [], motif: "Aucune catégorie de surface significative n'est renseignée pour évaluer cette voie." };

    const niveau = niveaux.every(n => n === "superieur") ? "superieur" : "base";
    return { eligible: true, niveau, manquants: [] };
}

function evaluerVoieCertification() {
    const env = profilExploitation.environnement;
    if (env.agricultureBiologique === true) return { eligible: true, niveau: "bio" };
    if (env.hveRenovee === true) return { eligible: true, niveau: "superieur" };
    if (env.certificationCE2Plus === true) return { eligible: true, niveau: "base" };

    const inconnu = [env.agricultureBiologique, env.hveRenovee, env.certificationCE2Plus].some(v => v === null);
    return { eligible: false, manquants: inconnu ? ["Certifications AB / HVE rénovée / CE2+"] : [] };
}

function evaluerVoieBiodiversite() {
    const ratio = profilExploitation.ecoregime.pourcentageIAEEtJacheresSAU;
    if (!estRenseigne(ratio)) return { eligible: false, manquants: ["Pourcentage d'IAE et jachères dans la SAU"] };
    if (nombreOuZero(ratio) >= 10) return { eligible: true, niveau: "superieur" };
    if (nombreOuZero(ratio) >= 7) return { eligible: true, niveau: "base" };
    return { eligible: false, manquants: [] };
}

function analyserEcoregime() {
    const id = "ecoregime";
    const actif = verifierActifPourAide(id);
    if (actif) return actif;

    if (!estRenseigne(profilExploitation.dpb.nombreDPBActives)) {
        return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], ["DPB activés"]);
    }
    if (!aDesDPBActives()) return nonRetenue(id, "L'accès à l'écorégime suppose de disposer de DPB activables.");
    if (!valeurPositive(profilExploitation.surfaces.surfaceAdmissible)) return nonRetenue(id, "Aucune surface admissible positive n'est renseignée.");

    const voies = [
        { id: "pratiques", ...evaluerVoiePratiques() },
        { id: "certification", ...evaluerVoieCertification() },
        { id: "biodiversite", ...evaluerVoieBiodiversite() }
    ];

    const eligibles = voies.filter(v => v.eligible);
    if (!eligibles.length) {
        const manquants = [...new Set(voies.flatMap(v => v.manquants || []))];
        if (manquants.length) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], manquants);
        return nonRetenue(id, "Aucune des trois voies d'accès ne semble atteindre le niveau de base avec les informations saisies.");
    }

    eligibles.sort((a, b) => niveauScore(b.niveau) - niveauScore(a.niveau));
    const meilleure = eligibles[0];

    const s = profilExploitation.surfaces;
    const env = profilExploitation.environnement;
    const taSignificatives = categorieSignificative(s.terresArables, s.surfaceAdmissible);
    const voieCompatibleHaies = eligibles.some(v => v.id === "pratiques" || v.id === "certification");
    const bonusHaies =
        voieCompatibleHaies &&
        nombreOuZero(s.pourcentageHaiesSAU) >= 6 &&
        (!taSignificatives || nombreOuZero(s.pourcentageHaiesTerresArables) >= 6) &&
        env.certificationGestionDurableHaies === true &&
        env.maecEntretienLigneux !== true;

    const raisons = [
        `Voie(s) initialement compatible(s) : ${eligibles.map(v => v.id).join(", ")}.`,
        `Niveau le plus élevé identifié : ${meilleure.niveau}.`
    ];
    if (bonusHaies) raisons.push("Les informations saisies sont compatibles avec le bonus haies.");

    return creerResultatAide(id, STATUT_AIDE.A_EXAMINER, raisons, [], [], {
        niveau: meilleure.niveau,
        voie: meilleure.id,
        voiesEligibles: eligibles.map(v => ({ voie: v.id, niveau: v.niveau })),
        bonusHaies
    });
}

    return creerResultatAide(id, STATUT_AIDE.A_EXAMINER, ["Zone de montagne ou haute montagne renseignée.", "Au moins 1 ha de cultures de vente renseigné."], [], verifications);
}

const TYPE_CALCUL = {
    CAMPAGNE_2026: "campagne_2026",
    PROFIL_EXPLOITATION: "profil_exploitation",
    REFERENCE_2024: "reference_2024",
    REFERENCE_2025: "reference_2025",
    PARTIEL: "partiel",
    NON_CALCULABLE: "non_calculable"
};

function arrondirMontant(montant) {
    if (!Number.isFinite(Number(montant))) return null;
    return Math.round(Number(montant) * 100) / 100;
}

function formaterMontant(montant) {
    if (montant === null || montant === undefined || !Number.isFinite(Number(montant))) return "—";
    return new Intl.NumberFormat("fr-FR", { style: "currency", currency: "EUR" }).format(Number(montant));
}

function creerCalcul({
    id,
    nom = null,
    montant = null,
    type = TYPE_CALCUL.NON_CALCULABLE,
    anneeReference = null,
    formule = null,
    donnees = {},
    avertissements = [],
    details = {}
}) {
    const aide = obtenirAide(id);
    const valeur = montant === null ? null : arrondirMontant(montant);
    return {
        id,
        nom: nom || (aide ? aide.nom : id),
        montant: valeur,
        montantFormate: formaterMontant(valeur),
        type,
        anneeReference,
        formule,
        donnees,
        avertissements,
        details
    };
}

function choisirTarif(montant2026, reference, anneeReference = 2024) {
    if (montant2026 !== null && montant2026 !== undefined) {
        return { tarif: Number(montant2026), type: TYPE_CALCUL.CAMPAGNE_2026, anneeReference: 2026 };
    }
    if (reference !== null && reference !== undefined) {
        return {
            tarif: Number(reference),
            type: anneeReference === 2025 ? TYPE_CALCUL.REFERENCE_2025 : TYPE_CALCUL.REFERENCE_2024,
            anneeReference
        };
    }
    return { tarif: null, type: TYPE_CALCUL.NON_CALCULABLE, anneeReference: null };
}

function orientationValide(orientation) {
    return orientation && orientation.statut === STATUT_AIDE.A_EXAMINER;
}

function calculNonEligible(orientation) {
    return creerCalcul({
        id: orientation.id,
        nom: orientation.nom,
        type: TYPE_CALCUL.NON_CALCULABLE,
        avertissements: orientation.informationsManquantes.length
            ? orientation.informationsManquantes
            : orientation.verifications
    });
}

function calculerAideBase() {
    const orientation = analyserAideBase();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);

    const surface = nombreOuZero(profilExploitation.surfaces.surfaceAdmissible);
    const dpbActifs = nombreOuZero(profilExploitation.dpb.nombreDPBActives);
    const nombrePrime = Math.min(surface, dpbActifs);
    const totalDeclare = profilExploitation.dpb.valeurTotaleDPB;
    const moyenne = profilExploitation.dpb.valeurMoyenneDPB;

    if (estRenseigne(totalDeclare) && Number(totalDeclare) >= 0) {
        return creerCalcul({
            id: "aide_base",
            montant: Number(totalDeclare),
            type: TYPE_CALCUL.PROFIL_EXPLOITATION,
            formule: "Valeur totale des DPB renseignée par l'utilisateur",
            donnees: { nombreDPBActives: dpbActifs, surfaceAdmissible: surface }
        });
    }

    if (estRenseigne(moyenne) && Number(moyenne) >= 0) {
        return creerCalcul({
            id: "aide_base",
            montant: nombrePrime * Number(moyenne),
            type: TYPE_CALCUL.PROFIL_EXPLOITATION,
            formule: "min(DPB activés, surface admissible) × valeur moyenne des DPB renseignée",
            donnees: { nombrePrime, valeurMoyenneDPB: Number(moyenne) }
        });
    }

    return creerCalcul({
        id: "aide_base",
        type: TYPE_CALCUL.NON_CALCULABLE,
        avertissements: ["Renseignez la valeur moyenne ou la valeur totale réelle des DPB. La moyenne nationale de référence n'est pas utilisée comme substitut à la valeur de l'exploitation."]
    });
}

function calculerAideRedistributive() {
    const orientation = analyserAideRedistributive();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);

    const p = PARAMETRES_PAC_2026.aideRedistributive;
    const t = choisirTarif(p.montant2026, p.reference2024, 2024);
    const surface = Math.min(nombreOuZero(profilExploitation.surfaces.surfaceAdmissible), p.hectaresMaximum);
    if (t.tarif === null) return creerCalcul({ id: "aide_redistributive", avertissements: ["Montant unitaire non disponible."] });

    return creerCalcul({
        id: "aide_redistributive",
        montant: surface * t.tarif,
        type: t.type,
        anneeReference: t.anneeReference,
        formule: `min(surface admissible, ${p.hectaresMaximum} ha) × tarif`,
        donnees: { surface, tarif: t.tarif },
        avertissements: profilExploitation.general.gaecTotal ? ["Calcul présenté hors application détaillée de la transparence GAEC."] : []
    });
}

function calculerACJA() {
    const orientation = analyserACJA();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);

    const p = PARAMETRES_PAC_2026.ACJA;
    const t = choisirTarif(p.montant2026, p.reference2024, 2024);
    if (t.tarif === null) return creerCalcul({ id: "acja", avertissements: ["Forfait annuel non disponible."] });

    let multiplicateur = 1;
    if (profilExploitation.general.gaecTotal === true && nombreOuZero(profilExploitation.installation.nombreAssociesJAEligibles) > 0) {
        multiplicateur = nombreOuZero(profilExploitation.installation.nombreAssociesJAEligibles);
    }

    return creerCalcul({
        id: "acja",
        montant: t.tarif * multiplicateur,
        type: profilExploitation.general.gaecTotal && multiplicateur > 1 ? TYPE_CALCUL.PARTIEL : t.type,
        anneeReference: t.anneeReference,
        formule: "Forfait annuel × nombre de bénéficiaires JA pris en compte",
        donnees: { forfait: t.tarif, multiplicateur },
        avertissements: profilExploitation.general.gaecTotal ? ["La transparence GAEC et le nombre d'associés JA éligibles doivent être confirmés administrativement."] : []
    });
}

function calculerEcoregime() {
    const orientation = analyserEcoregime();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);

    const p = PARAMETRES_PAC_2026.ecoregime;
    const niveau = orientation.details.niveau;
    let t;
    if (niveau === "bio") t = choisirTarif(p.niveauBio2026, p.reference2024.niveauBio, 2024);
    else if (niveau === "superieur") t = choisirTarif(p.niveauSuperieur2026, p.reference2024.niveauSuperieur, 2024);
    else t = choisirTarif(p.niveauBase2026, p.reference2024.niveauBase, 2024);

    const surface = nombreOuZero(profilExploitation.surfaces.surfaceAdmissible);
    const avertissements = [];

    if (niveau === "base" && t.tarif === null) {
        avertissements.push("Le guide comporte deux montants 2024 différents pour le niveau de base (48,23 et 48,35 €/ha). Aucun n'est choisi automatiquement.");
        return creerCalcul({ id: "ecoregime", type: TYPE_CALCUL.NON_CALCULABLE, avertissements, details: orientation.details });
    }

    let montant = surface * t.tarif;
    let type = t.type;
    let anneeReference = t.anneeReference;

    if (orientation.details.bonusHaies) {
        const bonus = choisirTarif(p.bonusHaies2026, p.reference2024.bonusHaies, 2024);
        if (bonus.tarif !== null) {
            montant += surface * bonus.tarif;
            if (bonus.type !== type) type = TYPE_CALCUL.PARTIEL;
            avertissements.push(`Bonus haies ajouté avec la référence ${bonus.anneeReference}.`);
        } else {
            avertissements.push("Bonus haies identifié mais tarif annuel non disponible.");
        }
    }

    return creerCalcul({
        id: "ecoregime",
        montant,
        type,
        anneeReference,
        formule: "Surface admissible × tarif du niveau identifié, avec éventuel bonus haies",
        donnees: { surface, niveau, voie: orientation.details.voie, tarif: t.tarif },
        avertissements,
        details: orientation.details
    });
}

function calculerBovinsGenerique(id, orientation, param) {
    if (!orientationValide(orientation)) return calculNonEligible(orientation);
    const b = profilExploitation.bovins;
    const surfaceFourragere = nombreOuZero(profilExploitation.surfaces.surfaceFourragere);
    const plafondSurface = Math.max(40, 1.4 * surfaceFourragere);
    const plafondGlobal = Math.min(120, plafondSurface);
    const sup = Math.min(nombreOuZero(b.ugbNiveauSuperieur), plafondGlobal);
    const reste = Math.max(0, plafondGlobal - sup);
    const base = Math.min(nombreOuZero(b.ugbNiveauBase), 40, reste);
    const tSup = choisirTarif(param.superieur.montant2026, param.superieur.reference2024, 2024);
    const tBase = choisirTarif(param.base.montant2026, param.base.reference2024, 2024);

    const montant = sup * nombreOuZero(tSup.tarif) + base * nombreOuZero(tBase.tarif);
    return creerCalcul({
        id,
        montant,
        type: (tSup.type === tBase.type) ? tSup.type : TYPE_CALCUL.PARTIEL,
        anneeReference: tSup.anneeReference === tBase.anneeReference ? tSup.anneeReference : null,
        formule: "UGB niveau supérieur × tarif supérieur + UGB niveau de base × tarif de base, après plafonds simplifiés",
        donnees: { ugbSuperieurRetenues: sup, ugbBaseRetenues: base, plafondGlobal, surfaceFourragere },
        avertissements: ["Calcul indicatif : le classement précis des animaux et les plafonds doivent être vérifiés administrativement."]
    });
}

function calculerOvins() {
    const orientation = analyserOvins();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);
    const o = profilExploitation.ovins;
    const p = PARAMETRES_PAC_2026.aidesCouplees.animales.ovinsHexagone;
    const baseTarif = choisirTarif(p.base.montant2026, p.base.reference2024, 2024);
    const majTarif = choisirTarif(p.majoration500Premieres.montant2026, p.majoration500Premieres.reference2024, 2024);
    const nouvTarif = choisirTarif(p.nouveauProducteur.montant2026, p.nouveauProducteur.reference2024, 2024);

    const demandees = nombreOuZero(o.nombreBrebis);
    let primees = demandees;
    if (o.nouveauProducteur !== true) {
        const presentesN1 = nombreOuZero(o.nombreBrebisPresentes1JanvierNMoins1);
        const agneaux = nombreOuZero(o.nombreAgneauxVendusNMoins1);
        const ratio = presentesN1 > 0 ? agneaux / presentesN1 : 0;
        if (ratio < 0.5 && presentesN1 > 0) primees = Math.min(demandees, agneaux / 0.5);
    }

    const majorables = Math.min(primees, 500);
    let montant = primees * baseTarif.tarif + majorables * majTarif.tarif;
    if (o.nouveauProducteur === true) montant += primees * nouvTarif.tarif;

    return creerCalcul({
        id: "ovins_hexagone",
        montant,
        type: TYPE_CALCUL.REFERENCE_2024,
        anneeReference: 2024,
        formule: "Brebis primables × base + majoration des 500 premières + éventuel complément nouveau producteur",
        donnees: { brebisDemandees: demandees, brebisPrimees: arrondirMontant(primees), brebisMajorees: arrondirMontant(majorables) },
        avertissements: ["Le guide ne précise pas ici les règles d'arrondi du nombre de brebis après réduction de productivité ; le résultat est indicatif.", ...(profilExploitation.general.gaecTotal ? ["Transparence GAEC à vérifier pour la majoration des 500 premières brebis."] : [])]
    });
}

function calculerICHNAnimale() {
    const orientation = analyserICHNAnimale();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);
    const i = profilExploitation.ichn;
    const p = PARAMETRES_PAC_2026.ICHN.animale;
    const surface = Math.min(nombreOuZero(i.surfacesFourrageresEligibles), p.surfaceMaximum);
    const partFixe = surface * p.montantBaseParHa;

    if (!estRenseigne(i.montantVariableSousZone) || !estRenseigne(i.coefficientModulationChargement)) {
        return creerCalcul({
            id: "ichn_animale",
            type: TYPE_CALCUL.NON_CALCULABLE,
            formule: "Part fixe + part variable locale, modulées selon le chargement",
            donnees: { surfaceRetenue: surface, partFixeTheorique: arrondirMontant(partFixe) },
            avertissements: [
                `Part fixe théorique avant modulation : ${formaterMontant(partFixe)}.`,
                "Le montant variable de sous-zone et le coefficient de chargement sont fixés localement ; le montant final n'est donc pas calculé."
            ]
        });
    }

    const tranche1 = Math.min(surface, 25);
    const tranche2 = Math.max(0, Math.min(surface, 50) - 25);
    const variable = tranche1 * Number(i.montantVariableSousZone) + tranche2 * Number(i.montantVariableSousZone) * p.coefficientDeuxiemeTranche;
    const montant = (partFixe + variable) * Number(i.coefficientModulationChargement);

    return creerCalcul({
        id: "ichn_animale",
        montant,
        type: TYPE_CALCUL.PARTIEL,
        formule: "(part fixe + part variable locale) × coefficient de modulation",
        donnees: { surface, partFixe, partVariable: variable, coefficient: Number(i.coefficientModulationChargement) },
        avertissements: ["Calcul à confirmer avec l'arrêté préfectoral applicable et les règles de revenu / chargement."]
    });
}

function calculerICHNVegetale() {
    const orientation = analyserICHNVegetale();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);
    const i = profilExploitation.ichn;
    const p = PARAMETRES_PAC_2026.ICHN.vegetale;
    if (i.zoneSeche === null) return creerCalcul({ id: "ichn_vegetale", avertissements: ["Précisez si les surfaces sont classées en zone sèche."] });

    const surface = Math.min(nombreOuZero(i.surfacesVegetalesEligibles), p.surfaceMaximum);
    const tarif = i.zoneSeche ? p.montantZoneSecheParHa : p.montantAutreZoneParHa;
    const t1 = Math.min(surface, p.premiereTranche);
    const t2 = Math.max(0, surface - p.premiereTranche);
    const montantAvantReduction = t1 * tarif + t2 * tarif * p.coefficientDeuxiemeTranche;

    if (nombreOuZero(i.pourcentageSAUZoneDefavorisee) < 80 || nombreOuZero(i.partRevenuAgricolePourcent) < 50) {
        return creerCalcul({
            id: "ichn_vegetale",
            type: TYPE_CALCUL.NON_CALCULABLE,
            donnees: { surface, tarif, montantAvantReduction: arrondirMontant(montantAvantReduction) },
            avertissements: [`Montant avant réduction éventuelle : ${formaterMontant(montantAvantReduction)}.`, "Le guide signale une réduction/plafonnement mais ne fournit pas ici le coefficient applicable à votre situation."]
        });
    }

    return creerCalcul({
        id: "ichn_vegetale",
        montant: montantAvantReduction,
        type: profilExploitation.general.gaecTotal ? TYPE_CALCUL.PARTIEL : TYPE_CALCUL.CAMPAGNE_2026,
        anneeReference: 2026,
        formule: "25 premiers ha au tarif plein + hectares 25 à 50 à 2/3 du tarif",
        donnees: { surface, tarif, zoneSeche: i.zoneSeche },
        avertissements: profilExploitation.general.gaecTotal ? ["La transparence GAEC doit être appliquée au plafond de surface."] : []
    });
}

function calculerSyntheseFinanciere() {
    const calculs = calculerToutesLesAides().filter(c => c.montant !== null);
    const totaux = {
        campagne2026: 0,
        profilExploitation: 0,
        reference2024: 0,
        reference2025: 0,
        partiel: 0
    };

    calculs.forEach(c => {
        if (c.type === TYPE_CALCUL.CAMPAGNE_2026) totaux.campagne2026 += c.montant;
        else if (c.type === TYPE_CALCUL.PROFIL_EXPLOITATION) totaux.profilExploitation += c.montant;
        else if (c.type === TYPE_CALCUL.REFERENCE_2024) totaux.reference2024 += c.montant;
        else if (c.type === TYPE_CALCUL.REFERENCE_2025) totaux.reference2025 += c.montant;
        else if (c.type === TYPE_CALCUL.PARTIEL) totaux.partiel += c.montant;
    });

    Object.keys(totaux).forEach(cle => { totaux[cle] = arrondirMontant(totaux[cle]); });
    return totaux;
}

function convertirValeur(valeur, type) {
    if (valeur === "") return null;
    if (valeur === "true") return true;
    if (valeur === "false") return false;
    if (type === "number") {
        const n = Number(valeur);
        return Number.isFinite(n) ? n : null;
    }
    return valeur;
}

function creerChampQuestion(question) {
    const valeur = lireProfil(question.chemin);
    const wrapper = document.createElement("div");
    wrapper.className = "question-item";

    const label = document.createElement("label");
    label.className = "question-label";
    label.textContent = question.label;
    wrapper.appendChild(label);

    if (question.type === "yesno") {
        const groupe = document.createElement("div");
        groupe.className = "yesno-group";
        [
            { texte: "Oui", valeur: true },
            { texte: "Non", valeur: false }
        ].forEach(option => {
            const bouton = document.createElement("button");
            bouton.type = "button";
            bouton.className = "answer-button" + (valeur === option.valeur ? " selected" : "");
            bouton.textContent = option.texte;
            bouton.addEventListener("click", () => {
                modifierProfil(question.chemin, option.valeur);
                sauvegarderProfilLocalement();
                afficherEtape();
            });
            groupe.appendChild(bouton);
        });
        wrapper.appendChild(groupe);
    } else if (question.type === "select") {
        const select = document.createElement("select");
        (question.options || []).forEach(option => {
            const opt = document.createElement("option");
            opt.value = option.valeur;
            opt.textContent = option.texte;
            if (String(valeur ?? "") === String(option.valeur)) opt.selected = true;
            select.appendChild(opt);
        });
        select.addEventListener("change", event => {
            modifierProfil(question.chemin, convertirValeur(event.target.value, "select"));
            sauvegarderProfilLocalement();
            if (question.refreshOnChange) afficherEtape();
        });
        wrapper.appendChild(select);
    } else {
        const ligne = document.createElement("div");
        ligne.className = "input-with-suffix";
        const input = document.createElement("input");
        input.type = question.type;
        input.value = valeur ?? "";
        if (question.placeholder) input.placeholder = question.placeholder;
        if (question.min !== undefined) input.min = question.min;
        if (question.max !== undefined) input.max = question.max;
        if (question.step !== undefined) input.step = question.step;
        input.addEventListener("change", event => {
            modifierProfil(question.chemin, convertirValeur(event.target.value, question.type));
            sauvegarderProfilLocalement();
            if (question.refreshOnChange) afficherEtape();
        });
        ligne.appendChild(input);
        if (question.suffixe) {
            const suffixe = document.createElement("span");
            suffixe.className = "input-suffix";
            suffixe.textContent = question.suffixe;
            ligne.appendChild(suffixe);
        }
        wrapper.appendChild(ligne);
    }

    if (question.aide) {
        const aide = document.createElement("p");
        aide.className = "question-help";
        aide.textContent = question.aide;
        wrapper.appendChild(aide);
    }

    return wrapper;
}

function afficherEtape() {
    const etape = ETAPES_QUESTIONNAIRE[etapeQuestionnaire];
    document.getElementById("question-title").textContent = etape.titre;
    document.getElementById("question-subtitle").textContent = etape.sousTitre;

    const zone = document.getElementById("questions-container");
    zone.innerHTML = "";
    etape.questions.forEach(question => {
        if (question.afficherSi && !question.afficherSi()) return;
        zone.appendChild(creerChampQuestion(question));
    });

    const pourcentage = ((etapeQuestionnaire + 1) / ETAPES_QUESTIONNAIRE.length) * 100;
    document.getElementById("progress-bar").style.width = `${pourcentage}%`;
    document.getElementById("step-counter").textContent = `Étape ${etapeQuestionnaire + 1} sur ${ETAPES_QUESTIONNAIRE.length}`;
    document.getElementById("previous-button").style.display = etapeQuestionnaire === 0 ? "none" : "inline-flex";
    document.getElementById("next-button").textContent = etapeQuestionnaire === ETAPES_QUESTIONNAIRE.length - 1 ? "Analyser mon exploitation" : "Continuer";
    window.scrollTo({ top: 0, behavior: "smooth" });
}

function etapeSuivante() {
    sauvegarderProfilLocalement();
    if (etapeQuestionnaire < ETAPES_QUESTIONNAIRE.length - 1) {
        etapeQuestionnaire += 1;
        afficherEtape();
    } else {
        afficherResultatsQuestionnaire();
    }
}

function etapePrecedente() {
    if (etapeQuestionnaire > 0) {
        etapeQuestionnaire -= 1;
        afficherEtape();
    }
}

        campagne_2026: "Paramètre 2026 / règle du guide 2026",
        profil_exploitation: "Donnée réelle de l'exploitation",
        reference_2024: "Référence 2024",
        reference_2025: "Référence 2025",
        partiel: "Calcul partiel / à confirmer",
        non_calculable: "Non calculable avec les données disponibles"
    };
    return libelles[type] || type;
}

function creerCarteResultat(resultat, calcul) {
    const carte = document.createElement("article");
    carte.className = `result-card ${resultat.statut === STATUT_AIDE.A_EXAMINER ? "success" : "warning"}`;

    const montant = calcul && calcul.montant !== null
        ? `<div class="result-amount">${calcul.montantFormate}</div><div class="result-reference">${libelleTypeCalcul(calcul.type)}</div>`
        : `<div class="result-reference">${calcul ? libelleTypeCalcul(calcul.type) : ""}</div>`;

    const raisons = resultat.raisons.length
        ? `<div class="result-details"><strong>Pourquoi ?</strong>${resultat.raisons.map(r => `<p>✓ ${r}</p>`).join("")}</div>`
        : "";

    const manquants = resultat.informationsManquantes.length
        ? `<div class="result-details"><strong>À compléter</strong>${resultat.informationsManquantes.map(r => `<p>• ${r}</p>`).join("")}</div>`
        : "";

    const avertissements = [
        ...(resultat.verifications || []),
        ...((calcul && calcul.avertissements) || [])
    ];
    const avertissementsHtml = avertissements.length
        ? `<div class="result-warnings">${[...new Set(avertissements)].map(a => `<p>${a}</p>`).join("")}</div>`
        : "";

    carte.innerHTML = `
        <div class="result-status">${resultat.statut === STATUT_AIDE.A_EXAMINER ? "À EXAMINER" : "INFORMATIONS À COMPLÉTER"}</div>
        <h3>${resultat.nom}</h3>
        ${montant}
        ${raisons}
        ${manquants}
        ${avertissementsHtml}
        <div class="result-source">Source : ${resultat.source}</div>
    `;
    return carte;
}

function afficherSynthese(zone) {
    const synthese = calculerSyntheseFinanciere();
    const valeurs = [
        ["Données exploitation", synthese.profilExploitation],
        ["Calculs / paramètres 2026", synthese.campagne2026],
        ["Références 2024", synthese.reference2024],
        ["Références 2025", synthese.reference2025],
        ["Calculs partiels", synthese.partiel]
    ].filter(([, montant]) => montant > 0);

    if (!valeurs.length) return;
    const grille = document.createElement("div");
    grille.className = "summary-grid";
    valeurs.forEach(([libelle, montant]) => {
        const bloc = document.createElement("div");
        bloc.className = "summary-card";
        bloc.innerHTML = `<span>${libelle}</span><strong>${formaterMontant(montant)}</strong>`;
        grille.appendChild(bloc);
    });
    zone.appendChild(grille);

    const note = document.createElement("p");
    note.className = "summary-note";
    note.textContent = "Les totaux sont séparés par nature : ils ne doivent pas être additionnés comme s'il s'agissait tous de montants définitifs 2026.";
    zone.appendChild(note);
}

function afficherResultatsQuestionnaire() {
    const questionnaire = document.getElementById("questionnaire-card");
    const zone = document.getElementById("results");
    questionnaire.style.display = "none";
    zone.style.display = "block";
    zone.innerHTML = "";

    const orientations = analyserToutesLesAides();
    const calculs = calculerToutesLesAides();
    const aExaminer = orientations.filter(r => r.statut === STATUT_AIDE.A_EXAMINER);
    const incompletes = orientations.filter(r => r.statut === STATUT_AIDE.INFORMATIONS_MANQUANTES);
    const nonRetenues = orientations.filter(r => r.statut === STATUT_AIDE.NON_RETENUE);

    const header = document.createElement("div");
    header.className = "results-header";
    header.innerHTML = `
        <span class="eyebrow">ORIENTATION PAC 2026</span>
        <h2>${aExaminer.length} aide${aExaminer.length > 1 ? "s" : ""} à examiner</h2>
        <p>Orientation fondée sur les informations saisies et le guide PAC utilisé. Elle ne constitue pas une décision administrative d'éligibilité.</p>
    `;
    zone.appendChild(header);

    if (profilExploitation.general.territoire === "Outre-mer") {
        const alerte = document.createElement("div");
        alerte.className = "notice warning-notice";
        alerte.textContent = "Les modalités spécifiques Outre-mer ne sont pas détaillées dans cette version du moteur. Les résultats nationaux affichés doivent être considérés comme incomplets.";
        zone.appendChild(alerte);
    }

    afficherSynthese(zone);

    aExaminer.forEach(resultat => {
        const calcul = calculs.find(c => c.id === resultat.id);
        zone.appendChild(creerCarteResultat(resultat, calcul));
    });

    if (incompletes.length) {
        const titre = document.createElement("h3");
        titre.className = "results-section-title";
        titre.textContent = "Informations utiles à compléter";
        zone.appendChild(titre);
        incompletes.forEach(resultat => {
            const calcul = calculs.find(c => c.id === resultat.id);
            zone.appendChild(creerCarteResultat(resultat, calcul));
        });
    }

    if (nonRetenues.length) {
        const details = document.createElement("details");
        details.className = "not-retained";
        const summary = document.createElement("summary");
        summary.textContent = `${nonRetenues.length} aide(s) non retenue(s) avec les informations actuelles`;
        details.appendChild(summary);
        nonRetenues.forEach(resultat => {
            const ligne = document.createElement("div");
            ligne.className = "not-retained-item";
            ligne.innerHTML = `<strong>${resultat.nom}</strong>${resultat.verifications.length ? `<span>${resultat.verifications[0]}</span>` : ""}`;
            details.appendChild(ligne);
        });
        zone.appendChild(details);
    }

    const actions = document.createElement("div");
    actions.className = "results-actions";
    actions.innerHTML = `
        <button type="button" class="secondary-button" id="edit-results-button">Modifier mes réponses</button>
        <button type="button" class="danger-button" id="reset-results-button">Recommencer à zéro</button>
    `;
    zone.appendChild(actions);

    document.getElementById("edit-results-button").addEventListener("click", recommencerQuestionnaire);
    document.getElementById("reset-results-button").addEventListener("click", () => {
        reinitialiserProfil();
        etapeQuestionnaire = 0;
        zone.style.display = "none";
        questionnaire.style.display = "block";
        afficherEtape();
    });

    window.scrollTo({ top: 0, behavior: "smooth" });
}

function recommencerQuestionnaire() {
    document.getElementById("results").style.display = "none";
    document.getElementById("questionnaire-card").style.display = "block";
    etapeQuestionnaire = 0;
    afficherEtape();
}

function initialiserQuestionnaire() {
    chargerProfilLocal();
    afficherEtape();
    document.getElementById("next-button").addEventListener("click", etapeSuivante);
    document.getElementById("previous-button").addEventListener("click", etapePrecedente);
    document.getElementById("reset-button").addEventListener("click", () => {
        if (!confirm("Effacer toutes les réponses enregistrées dans ce navigateur ?")) return;
        reinitialiserProfil();
        etapeQuestionnaire = 0;
        document.getElementById("results").style.display = "none";
        document.getElementById("questionnaire-card").style.display = "block";
        afficherEtape();
    });
}
