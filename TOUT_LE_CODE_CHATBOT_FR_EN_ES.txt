    const language = lang || (typeof getChatbotLanguage === "function" ? getChatbotLanguage() : "fr");
    const dict = CHATBOT_I18N[language] || CHATBOT_I18N.fr;
    let value = dict[key] ?? CHATBOT_I18N.fr[key] ?? key;
    Object.entries(variables).forEach(([name, replacement]) => {
        value = value.replaceAll(`{${name}}`, String(replacement));
    });
    return value;
}

function cloner(objet) {
    return JSON.parse(JSON.stringify(objet));
}

function fusionnerProfondement(cible, source) {
    if (!source || typeof source !== "object") return cible;
    Object.keys(source).forEach(cle => {
        const valeur = source[cle];
        if (
            valeur &&
            typeof valeur === "object" &&
            !Array.isArray(valeur) &&
            cible[cle] &&
            typeof cible[cle] === "object"
        ) {
            fusionnerProfondement(cible[cle], valeur);
        } else {
            cible[cle] = valeur;
        }
    });
    return cible;
}

function nombreOuZero(valeur) {
    const nombre = Number(valeur);
    return Number.isFinite(nombre) ? nombre : 0;
}

function estRenseigne(valeur) {
    return valeur !== null && valeur !== undefined && valeur !== "";

function normaliserTexte(texte) {
    return String(texte || "")
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase()
        .trim();
}

function lireChemin(objet, chemin) {
    return chemin.split(".").reduce((courant, cle) => {
        if (courant === null || courant === undefined) return undefined;
        return courant[cle];
    }, objet);
}

function ecrireChemin(objet, chemin, valeur) {
    const morceaux = chemin.split(".");
    const derniere = morceaux.pop();
    let courant = objet;
    morceaux.forEach(cle => {
        if (!courant[cle] || typeof courant[cle] !== "object") courant[cle] = {};
        courant = courant[cle];
    });
    courant[derniere] = valeur;
}

function valeurConnue(v) {
    return v !== null && v !== undefined && v !== "";
}

function ratio(num, den) {
    const d = nombreOuZero(den);
    if (d <= 0) return null;
    return nombreOuZero(num) / d;
}

function premierBareme(ratioValeur, bareme) {
    if (ratioValeur === null) return 0;
    const ligne = bareme.find(item => ratioValeur >= item.seuil);
    return ligne ? ligne.points : 0;
}

function calculerAgriculteurActif() {
    const g = profilExploitation.general;
    const s = profilExploitation.statutActif;
    const statut = g.statutJuridique;
    const manquants = [];
    const raisons = [];

    if (g.territoire === "Outre-mer") {
        return {
            statut: null,
            manquants: ["Règles spécifiques Outre-mer non détaillées dans ce moteur"],
            raisons: ["Le guide prévoit une définition spécifique de l'agriculteur actif dans les territoires ultramarins."]
        };
    }

    if (!statut) {
        return { statut: null, manquants: ["Statut juridique"], raisons: [] };
    }

    if (statut === "indivision") {
        return { statut: false, manquants: [], raisons: ["Les indivisions ne sont pas considérées comme agriculteurs actifs, hors cas successoraux spécifiques."] };
    }

    if (statut === "individuel") {
        if (s.affilieATEXA === null) manquants.push("Affiliation ATEXA");
        if (!valeurConnue(s.ageExploitant)) manquants.push("Âge de l'exploitant");
        if (manquants.length) return { statut: null, manquants, raisons };
        if (s.affilieATEXA !== true) return { statut: false, manquants: [], raisons: ["Affiliation ATEXA non renseignée comme satisfaite."] };

        if (nombreOuZero(s.ageExploitant) > 67) {
            if (s.retraiteLiquideeApres67 === null) return { statut: null, manquants: ["Situation de retraite après 67 ans"], raisons };
            if (s.retraiteLiquideeApres67 === true) return { statut: false, manquants: [], raisons: ["Après 67 ans, les droits à la retraite sont renseignés comme liquidés."] };
        }

        raisons.push("Affiliation ATEXA renseignée.");
        raisons.push("Critère d'âge / retraite compatible avec la définition du guide.");
        return { statut: true, manquants: [], raisons };
    }

    const societesCourantes = ["GAEC", "EARL", "SCEA", "SAS", "SARL", "SA"];
    if (societesCourantes.includes(statut)) {
        if (s.sansAssocieATEXA === false || s.associeAffilieATEXA === true) {
            if (s.associeAffilieATEXA === null) manquants.push("Présence d'au moins un associé affilié à l'ATEXA");
            if (s.associeRespecteCritereAgeRetraite === null) manquants.push("Critère âge / retraite de l'associé");
            if (manquants.length) return { statut: null, manquants, raisons };
            if (s.associeAffilieATEXA !== true || s.associeRespecteCritereAgeRetraite !== true) {
                return { statut: false, manquants: [], raisons: ["Aucun associé ne remplit l'ensemble des conditions renseignées pour une personne physique."] };
            }
            return { statut: true, manquants: [], raisons: ["Au moins un associé remplit les conditions renseignées d'une personne physique active."] };
        }

        if (s.sansAssocieATEXA === null) {
            return { statut: null, manquants: ["Présence ou non d'un associé cotisant à l'ATEXA"], raisons };
        }

        if (s.sansAssocieATEXA === true) {
            const champs = [
                [s.activiteAgricoleSociete, "Activité agricole de la société"],
                [s.dirigeantsAffiliesATMPAgricole, "Affiliation AT/MP agricole des dirigeants associés"],
                [s.dirigeantsRespectentRetraite, "Situation retraite des dirigeants associés"],
                [s.dirigeantsDetiennent5Pourcent, "Détention d'au moins 5 % des parts par les dirigeants"]
            ];
            champs.forEach(([v, lib]) => { if (v === null) manquants.push(lib); });
            if (manquants.length) return { statut: null, manquants, raisons };
            if (champs.every(([v]) => v === true)) {
                return { statut: true, manquants: [], raisons: ["Les critères renseignés pour une société sans associé ATEXA sont satisfaits."] };
            }
            return { statut: false, manquants: [], raisons: ["Au moins un critère renseigné pour les sociétés sans associé ATEXA n'est pas satisfait."] };
        }
    }

    if (["association", "fondation", "public", "cooperative", "autre"].includes(statut)) {
        if (s.autrePersonneMoraleActiviteAgricoleStatutaire === null) {
            return { statut: null, manquants: ["Activité agricole statutaire / propre à la structure"], raisons };
        }
        return s.autrePersonneMoraleActiviteAgricoleStatutaire === true
            ? { statut: true, manquants: [], raisons: ["La structure est renseignée comme exerçant l'activité agricole prévue par le guide."] }
            : { statut: false, manquants: [], raisons: ["L'activité agricole nécessaire n'est pas renseignée comme satisfaite."] };
    }

    return { statut: null, manquants: ["Forme juridique à vérifier"], raisons };
}

function calculerSommeAssolementTA() {

        a.plantesSarclees,

        a.oleagineuxPrintemps,
        a.autresCultures
    ].reduce((total, v) => total + nombreOuZero(v), 0);
}

function assolementTASuffisammentRenseigne() {
    const ta = nombreOuZero(profilExploitation.surfaces.terresArables);
    if (ta <= 0) return true;
    const a = profilExploitation.assolement;
    const champs = [a.prairieTemporaire, a.jachereTotale, a.legumineuses, a.cerealesHiver, a.cerealesPrintemps, a.plantesSarclees, a.oleagineuxHiver, a.oleagineuxPrintemps, a.autresCultures];
    if (!champs.some(valeurConnue)) return false;
    const somme = calculerSommeAssolementTA();
    const tolerance = Math.max(0.5, ta * 0.05);
    return Math.abs(somme - ta) <= tolerance;
}

function calculerScoreDiversification() {
    const r = REFERENTIEL_PAC.ecoregime.diversification;
    const s = profilExploitation.surfaces;
    const a = profilExploitation.assolement;
    const ta = nombreOuZero(s.terresArables);
    const sau = nombreOuZero(s.sau);

    if (ta <= 0) {
        return { points: 0, complet: true, details: [], ecartAssolement: 0 };
    }

    const complet = assolementTASuffisammentRenseigne();
    const details = [];
    let points = 0;

    const ptEco = nombreOuZero(a.prairieTemporaire) + nombreOuZero(a.jachereEcoEligible);
    const ratioPT = ptEco / ta;
    const pPT = premierBareme(ratioPT, r.prairieTemporaireEtJachere);
    if (pPT > 0) details.push({ libelle: "Prairie temporaire / jachère écorégime", points: pPT, ratio: ratioPT });
    points += pPT;

    const legumes = nombreOuZero(a.legumineuses);
    const ratioLeg = legumes / ta;
    let pLeg = 0;
    if (ratioLeg >= 0.10) pLeg = 3;
    else if (ratioLeg >= 0.05 || legumes > r.seuilLegumineusesHaAlternative) pLeg = 2;
    if (pLeg > 0) details.push({ libelle: "Légumineuses", points: pLeg, ratio: ratioLeg });
    points += pLeg;

    const bloc = r.blocCerealesSarcleesOleagineux;
    const categoriesBloc = [
        ["Céréales d'hiver", nombreOuZero(a.cerealesHiver), bloc.cerealesHiver.seuil],
        ["Céréales de printemps", nombreOuZero(a.cerealesPrintemps), bloc.cerealesPrintemps.seuil],
        ["Plantes sarclées", nombreOuZero(a.plantesSarclees), bloc.plantesSarclees.seuil],
        ["Oléagineux d'hiver", nombreOuZero(a.oleagineuxHiver), bloc.oleagineuxHiver.seuil],
        ["Oléagineux de printemps", nombreOuZero(a.oleagineuxPrintemps), bloc.oleagineuxPrintemps.seuil]
    ];
    let pBloc = 0;
    categoriesBloc.forEach(([libelle, surface, seuil]) => {
        if (surface / ta >= seuil) {
            pBloc += 1;
            details.push({ libelle, points: 1, ratio: surface / ta });
        }
    });
    pBloc = Math.min(bloc.plafondPoints, pBloc);
    if (pBloc === 0) {
        const sommeBloc = categoriesBloc.reduce((t, [, surface]) => t + surface, 0);
        if (sommeBloc / ta >= bloc.secoursSiAucunSeuil.seuilSomme) {
            pBloc = bloc.secoursSiAucunSeuil.points;
            details.push({ libelle: "Ensemble céréales / sarclées / oléagineux", points: pBloc, ratio: sommeBloc / ta });
        }
    }
    points += pBloc;

    const autres = nombreOuZero(a.autresCultures);
    const ratioAutres = autres / ta;
    const pAutres = premierBareme(ratioAutres, r.autresCultures);
    if (pAutres > 0) details.push({ libelle: "Autres cultures", points: pAutres, ratio: ratioAutres });
    points += pAutres;

    if (sau > 0) {
        const ratioPP = nombreOuZero(s.prairiesPermanentes) / sau;
        const pPP = premierBareme(ratioPP, r.prairiePermanenteDansSAU);
        if (pPP > 0) details.push({ libelle: "Prairies permanentes dans la SAU", points: pPP, ratio: ratioPP });
        points += pPP;
    }

    if (ta < 10) {
        points += r.bonusTerresArablesMoins10Ha;
        details.push({ libelle: "Terres arables < 10 ha", points: r.bonusTerresArablesMoins10Ha, ratio: null });
    }

    return {
        points,
        complet,
        details,
        sommeAssolement: calculerSommeAssolementTA(),
        terresArables: ta,
        ecartAssolement: calculerSommeAssolementTA() - ta
    };
}

function categorieSignificative(surface) {
    const admissible = nombreOuZero(profilExploitation.surfaces.surfaceAdmissible);
    if (admissible <= 0) return false;
    return nombreOuZero(surface) / admissible >= REFERENTIEL_PAC.ecoregime.seuilCategorieSignificative;
}

function evaluerEcoregimePratiques() {
    const s = profilExploitation.surfaces;
    const p = profilExploitation.pratiques;
    const score = calculerScoreDiversification();
    const exigences = [];
    const manquants = [];
    let echec = null;

    if (nombreOuZero(s.surfaceAdmissible) <= 0) {
        return { eligible: false, niveau: null, manquants: ["Surface admissible"], exigences, score };
    }

    if (categorieSignificative(s.terresArables)) {
        if (!score.complet) manquants.push("Répartition complète des terres arables par catégories écorégime");
        else if (score.points < 4) echec = `Score de diversification calculé : ${score.points} point(s), inférieur au niveau de base.`;
        else exigences.push({ compartiment: "Terres arables", niveau: score.points >= 5 ? "superieur" : "base", valeur: `${score.points} points` });
    }

    if (!echec && categorieSignificative(s.prairiesPermanentes)) {
        if (!valeurConnue(p.surfacePPRespectantCritereEcoregime)) manquants.push("Surface de prairies permanentes respectant le critère de non-labour / traitement");
        else {
            const r = ratio(p.surfacePPRespectantCritereEcoregime, s.prairiesPermanentes);
            if (r < REFERENTIEL_PAC.ecoregime.pratiques.prairiePermanenteBase) echec = "Moins de 80 % des prairies permanentes respectent le critère écorégime.";
            else exigences.push({ compartiment: "Prairies permanentes", niveau: r >= REFERENTIEL_PAC.ecoregime.pratiques.prairiePermanenteSuperieur ? "superieur" : "base", valeur: `${Math.round(r * 1000) / 10} %` });
        }
    }

    if (!echec && categorieSignificative(s.culturesPermanentes)) {
        if (!valeurConnue(p.surfaceCPCouverteEcoregime)) manquants.push("Surface de cultures permanentes avec inter-rangs / surface couverts");
        else {
            const r = ratio(p.surfaceCPCouverteEcoregime, s.culturesPermanentes);
            if (r < REFERENTIEL_PAC.ecoregime.pratiques.culturesPermanentesBase) echec = "Moins de 75 % des cultures permanentes respectent le critère de couverture.";
            else exigences.push({ compartiment: "Cultures permanentes", niveau: r >= REFERENTIEL_PAC.ecoregime.pratiques.culturesPermanentesSuperieur ? "superieur" : "base", valeur: `${Math.round(r * 1000) / 10} %` });
        }
    }

    if (echec) return { eligible: false, niveau: null, manquants: [], motif: echec, exigences, score };
    if (manquants.length) return { eligible: false, niveau: null, manquants, exigences, score };
    if (!exigences.length) return { eligible: false, niveau: null, manquants: ["Répartition des surfaces"], exigences, score };

    const niveau = exigences.every(e => e.niveau === "superieur") ? "superieur" : "base";
    return { eligible: true, niveau, manquants: [], exigences, score };
}

function evaluerEcoregimeCertification() {
    const p = profilExploitation.pratiques;
    if (p.agricultureBiologiqueTouteExploitation === true) return { eligible: true, niveau: "bio", motif: "Agriculture biologique sur l'ensemble de l'exploitation" };
    if (p.hveRenovee === true) return { eligible: true, niveau: "superieur", motif: "Certification HVE rénovée" };
    if (p.certificationCE2Plus === true) return { eligible: true, niveau: "base", motif: "Certification CE2+" };

    const manquants = [];
    if (p.agricultureBiologiqueTouteExploitation === null) manquants.push("Certification AB de l'ensemble de l'exploitation");
    if (p.hveRenovee === null) manquants.push("Certification HVE rénovée");
    if (p.certificationCE2Plus === null) manquants.push("Certification CE2+");
    return { eligible: false, niveau: null, manquants };
}

function calculerIAE() {
    const s = profilExploitation.surfaces;
    const a = profilExploitation.assolement;
    const p = profilExploitation.pratiques;
    const eqHaie = REFERENTIEL_PAC.ecoregime.biodiversite.equivalenceHaieHaParMetre;
    const haiesEquivalentHa = nombreOuZero(p.longueurHaiesTotaleM) * eqHaie;
    const haiesTAEquivalentHa = nombreOuZero(p.longueurHaiesTerresArablesM) * eqHaie;
    const totalEquivalentHa = nombreOuZero(p.iaeAutresEquivalentHa) + nombreOuZero(a.jachereEcoEligible) + haiesEquivalentHa;
    const ratioSAU = ratio(totalEquivalentHa, s.sau);
    const ratioHaiesAdmissible = ratio(haiesEquivalentHa, s.surfaceAdmissible);
    const ratioHaiesTA = ratio(haiesTAEquivalentHa, s.terresArables);

    const donneesComplete =
        valeurConnue(p.iaeAutresEquivalentHa) &&
        valeurConnue(p.longueurHaiesTotaleM) &&
        (nombreOuZero(s.terresArables) <= 0 || valeurConnue(p.longueurHaiesTerresArablesM)) &&
        (nombreOuZero(s.terresArables) <= 0 || valeurConnue(a.jachereEcoEligible));

    return { haiesEquivalentHa, haiesTAEquivalentHa, totalEquivalentHa, ratioSAU, ratioHaiesAdmissible, ratioHaiesTA, donneesComplete };
}

function evaluerEcoregimeBiodiversite() {
    const iae = calculerIAE();
    if (nombreOuZero(profilExploitation.surfaces.sau) <= 0) return { eligible: false, niveau: null, manquants: ["SAU totale"], iae };
    if (!iae.donneesComplete) return { eligible: false, niveau: null, manquants: ["IAE, jachères éligibles et haies"], iae };
    if (iae.ratioSAU >= REFERENTIEL_PAC.ecoregime.biodiversite.superieur) return { eligible: true, niveau: "superieur", manquants: [], iae };
    if (iae.ratioSAU >= REFERENTIEL_PAC.ecoregime.biodiversite.base) return { eligible: true, niveau: "base", manquants: [], iae };
    return { eligible: false, niveau: null, manquants: [], motif: "Moins de 7 % d'IAE / jachères sur la SAU.", iae };
}

function evaluerBonusHaies(voies) {
    const p = profilExploitation.pratiques;
    const s = profilExploitation.surfaces;
    const iae = calculerIAE();
    const voieCompatible = voies.pratiques.eligible || voies.certification.eligible;
    const taApplicable = categorieSignificative(s.terresArables);
    const manquants = [];

    if (!voieCompatible) return { possible: false, manquants: [], motif: "Le bonus nécessite une voie pratiques ou certification." };
    if (p.certificationHaiesDurables === null) manquants.push("Certification de gestion durable des haies");
    if (p.maecEntretienLigneux === null) manquants.push("Engagement éventuel en MAEC entretien des ligneux");
    if (!iae.donneesComplete) manquants.push("Longueur de haies");
    if (manquants.length) return { possible: false, manquants };
    if (p.certificationHaiesDurables !== true) return { possible: false, manquants: [], motif: "Absence de certification de gestion durable des haies." };
    if (p.maecEntretienLigneux === true) return { possible: false, manquants: [], motif: "Incompatibilité avec la MAEC Biodiversité – entretien des ligneux." };
    if ((iae.ratioHaiesAdmissible || 0) < REFERENTIEL_PAC.ecoregime.bonusHaies.seuilSAU) return { possible: false, manquants: [], motif: "Moins de 6 % de haies sur la surface admissible." };
    if (taApplicable && (iae.ratioHaiesTA || 0) < REFERENTIEL_PAC.ecoregime.bonusHaies.seuilTerresArables) return { possible: false, manquants: [], motif: "Moins de 6 % de haies sur les terres arables." };
    return { possible: true, manquants: [], iae };
}

function calculerDiagnosticEcoregime() {
    const pratiques = evaluerEcoregimePratiques();
    const certification = evaluerEcoregimeCertification();
    const biodiversite = evaluerEcoregimeBiodiversite();
    const voies = { pratiques, certification, biodiversite };
    const bonusHaies = evaluerBonusHaies(voies);

    const rang = { base: 1, superieur: 2, bio: 3 };
    const eligibles = Object.entries(voies)
        .filter(([, v]) => v.eligible)
        .map(([voie, v]) => ({ voie, niveau: v.niveau, rang: rang[v.niveau] || 0 }));
    eligibles.sort((a, b) => b.rang - a.rang);

    return {
        voies,
        meilleureVoie: eligibles.length ? eligibles[0] : null,
        voiesEligibles: eligibles,
        bonusHaies,
        scoreDiversification: pratiques.score || calculerScoreDiversification()
    };
}

function calculerQualificationJA() {
    const i = profilExploitation.installation;
    const manquants = [];
    if (i.premiereInstallation === false) return { eligible: false, manquants: [], motif: "Pas de première installation récente." };
    if (i.premiereInstallation === null) manquants.push("Première installation");
    if (!valeurConnue(i.ageJA)) manquants.push("Âge du JA");
    if (!valeurConnue(i.anneeInstallation)) manquants.push("Année de première installation");
    if (!valeurConnue(profilExploitation.general.anneeDemande)) manquants.push("Année de campagne visée");
    if (i.diplomeAgricoleNiveau4OuPlus === null) manquants.push("Diplôme agricole niveau 4+");
    if (i.diplomeNiveau3OuPlus === null) manquants.push("Diplôme niveau 3+");
    if (!valeurConnue(i.experienceAgricoleMois)) manquants.push("Expérience agricole");
    if (manquants.length) return { eligible: false, manquants };

    if (nombreOuZero(i.ageJA) > 40) return { eligible: false, manquants: [], motif: "Âge supérieur à 40 ans." };
    const annee = nombreOuZero(profilExploitation.general.anneeDemande);
    const installation = nombreOuZero(i.anneeInstallation);
    if (installation > annee || installation < annee - 5) return { eligible: false, manquants: [], motif: "Installation hors de la fenêtre de première installation / cinq années civiles précédentes." };

    const exp = nombreOuZero(i.experienceAgricoleMois);
    const qualifie = i.diplomeAgricoleNiveau4OuPlus === true || exp >= 40 || (i.diplomeNiveau3OuPlus === true && exp >= 24);
    if (!qualifie) return { eligible: false, manquants: [], motif: "Diplôme / expérience insuffisants selon les critères du guide." };
    return { eligible: true, manquants: [], motif: "Critères d'âge, installation et qualification initialement compatibles." };
}

function calculerUGBBovines() {
    const b = profilExploitation.bovins;
    const males = nombreOuZero(b.malesPlus2Ans) + 0.6 * nombreOuZero(b.males16a24Mois);
    const femellesViande = nombreOuZero(b.femellesRaceViandePlus2Ans) + 0.6 * nombreOuZero(b.femellesRaceViande16a24Mois);
    const autresFemelles = nombreOuZero(b.autresFemellesPlus2Ans) + 0.6 * nombreOuZero(b.autresFemelles16a24Mois);
    const total = males + femellesViande + autresFemelles;

    const plafondMales = nombreOuZero(b.vachesEligibles);
    const plafondFemellesViande = 2 * nombreOuZero(b.veauxRaceViandeNesGardes90Jours);
    const niveauSuperieurBrut = Math.min(males, plafondMales) + Math.min(femellesViande, plafondFemellesViande);
    const niveauBaseBrut = Math.max(0, total - niveauSuperieurBrut);

    const sfConnue = valeurConnue(profilExploitation.surfaces.surfaceFourragere);
    const plafondSurface = sfConnue ? Math.max(40, 1.4 * nombreOuZero(profilExploitation.surfaces.surfaceFourragere)) : 120;
    const plafondGlobal = Math.min(120, plafondSurface);
    const niveauSuperieurRetenu = Math.min(niveauSuperieurBrut, plafondGlobal);
    const reste = Math.max(0, plafondGlobal - niveauSuperieurRetenu);
    const niveauBaseRetenu = Math.min(niveauBaseBrut, 40, reste);

    return {
        males,
        femellesViande,
        autresFemelles,
        total,
        niveauSuperieurBrut,
        niveauBaseBrut,
        niveauSuperieurRetenu,
        niveauBaseRetenu,
        plafondGlobal,
        surfaceFourragereConnue: sfConnue
    };
}

function calculerUGBTotalesDisponibles() {
    return calculerUGBBovines().total + nombreOuZero(profilExploitation.autresAnimaux.ugbDeclareesHorsBovins);
}

const STATUT_AIDE = {
    A_EXAMINER: "a_examiner",
    INFORMATIONS_MANQUANTES: "informations_manquantes",
    NON_RETENUE: "non_retenue"
};

function creerResultatAide(id, statut, raisons = [], informationsManquantes = [], verifications = [], details = {}) {
    const aide = obtenirAide(id);
    return {
        id,
        nom: aide ? aide.nom : id,
        categorie: aide ? aide.categorie : "",
        statut,
        raisons,
        informationsManquantes,
        verifications,
        details,
        source: aide ? aide.source : ""
    };
}

function nonRetenue(id, raison) {
    return creerResultatAide(id, STATUT_AIDE.NON_RETENUE, [], [], raison ? [raison] : []);
}

function verifierActifPourAide(id) {
    const actif = calculerAgriculteurActif();
    if (actif.statut === null) {
        return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], actif.manquants, actif.raisons);
    }
    if (actif.statut === false) {
        return nonRetenue(id, actif.raisons[0] || "Le critère d'agriculteur actif n'est pas rempli.");
    }
    return null;
}

function valeurPositive(v) {
    return nombreOuZero(v) > 0;
}

function analyserAideBase() {
    const id = "aide_base";
    const actif = verifierActifPourAide(id);
    if (actif) return actif;

    if (profilExploitation.dpb.detientDPB === false) return nonRetenue(id, "Aucun DPB n'est renseigné.");
    const manquants = [];
    if (profilExploitation.dpb.detientDPB === null) manquants.push("Détention de DPB");
    if (!estRenseigne(profilExploitation.dpb.nombreDPBActives)) manquants.push("Nombre de DPB activés");
    if (!estRenseigne(profilExploitation.surfaces.surfaceAdmissible)) manquants.push("Surface admissible");
    if (manquants.length) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], manquants);
    if (!aDesDPBActives() || nombreOuZero(profilExploitation.surfaces.surfaceAdmissible) <= 0) return nonRetenue(id, "Aucun DPB activé sur une surface admissible positive.");

    return creerResultatAide(id, STATUT_AIDE.A_EXAMINER, ["Des DPB activés et une surface admissible sont renseignés."], [], ["La valeur réelle des DPB détermine le montant de l'aide de base."]);
}

function analyserAideRedistributive() {
    const id = "aide_redistributive";
    const actif = verifierActifPourAide(id);
    if (actif) return actif;
    if (!estRenseigne(profilExploitation.dpb.nombreDPBActives)) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], ["Nombre de DPB activés"]);
    if (!aDesDPBActives()) return nonRetenue(id, "Au moins un DPB ou une fraction de DPB doit être activé.");
    if (!valeurPositive(profilExploitation.surfaces.surfaceAdmissible)) return nonRetenue(id, "Aucune surface admissible positive n'est renseignée.");
    return creerResultatAide(id, STATUT_AIDE.A_EXAMINER, ["Au moins un DPB est activé.", "Une surface admissible est renseignée."], [], profilExploitation.general.gaecTotal ? ["La transparence GAEC doit être appliquée."] : []);
}

function analyserACJA() {
    const id = "acja";
    const actif = verifierActifPourAide(id);
    if (actif) return actif;
    const ja = calculerQualificationJA();
    if (ja.manquants.length) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], ja.manquants);
    if (!ja.eligible) return nonRetenue(id, ja.motif);
    if (!estRenseigne(profilExploitation.dpb.nombreDPBActives)) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], ["DPB activés"]);
    if (!aDesDPBActives()) return nonRetenue(id, "Au moins un DPB ou une fraction de DPB doit être activé.");
    return creerResultatAide(id, STATUT_AIDE.A_EXAMINER, [ja.motif, "Au moins un DPB est activé."], [], profilExploitation.general.statutJuridique !== "individuel" ? ["Pour une société, confirmer administrativement l'associé qui confère le caractère JA."] : []);
}

function analyserEcoregime() {
    const id = "ecoregime";
    const actif = verifierActifPourAide(id);
    if (actif) return actif;

    if (!estRenseigne(profilExploitation.dpb.nombreDPBActives)) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], ["Nombre de DPB activés"]);
    if (!aDesDPBActives()) return nonRetenue(id, "L'accès à l'écorégime nécessite des DPB activés.");
    if (!valeurPositive(profilExploitation.surfaces.surfaceAdmissible)) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], ["Surface admissible"]);

    const diagnostic = calculerDiagnosticEcoregime();
    if (diagnostic.meilleureVoie) {
        const libelles = { pratiques: "pratiques", certification: "certification", biodiversite: "biodiversité" };
        const raisons = diagnostic.voiesEligibles.map(v => `Voie ${libelles[v.voie]} : niveau ${v.niveau}.`);
        if (diagnostic.bonusHaies.possible) raisons.push("Bonus haies potentiellement accessible par une voie compatible.");
        return creerResultatAide(id, STATUT_AIDE.A_EXAMINER, raisons, [], [], {
            niveau: diagnostic.meilleureVoie.niveau,
            voie: diagnostic.meilleureVoie.voie,
            voiesEligibles: diagnostic.voiesEligibles,
            bonusHaies: diagnostic.bonusHaies,
            scoreDiversification: diagnostic.scoreDiversification,
            diagnosticComplet: diagnostic
        });
    }

    const manquants = [
        ...(diagnostic.voies.pratiques.manquants || []),
        ...(diagnostic.voies.certification.manquants || []),
        ...(diagnostic.voies.biodiversite.manquants || [])
    ];
    const uniques = [...new Set(manquants)];
    if (uniques.length) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], uniques, [], { diagnosticComplet: diagnostic });

    return nonRetenue(id, "Aucune des trois voies de l'écorégime n'atteint le niveau de base avec les données renseignées.");
}

function analyserPommeTerreFeculiere() {

function analyserICHNAnimale() {
    const id = "ichn_animale";
    const i = profilExploitation.ichn;
    if (i.zoneEligible === false) return nonRetenue(id, "Aucune surface en zone défavorisée ICHN n'est renseignée.");
    if (!valeurPositive(i.surfacesFourrageresEligibles) && calculerUGBTotalesDisponibles() <= 0) return nonRetenue(id, "Aucune situation ICHN animale n'est renseignée.");
    const actif = verifierActifPourAide(id); if (actif) return actif;
    const manquants = [];
    if (i.zoneEligible === null) manquants.push("Présence de surfaces en zone ICHN");
    if (!i.typeZone) manquants.push("Type de zone ICHN");
    if (!estRenseigne(i.surfacesFourrageresEligibles)) manquants.push("Surface fourragère en zone défavorisée");
    if (!estRenseigne(i.pourcentageSAUZoneDefavorisee)) manquants.push("Part de SAU en zone défavorisée");
    if (!estRenseigne(i.partRevenuAgricolePourcent)) manquants.push("Part du revenu agricole");
    const ugb = calculerUGBTotalesDisponibles();
    if (ugb < REFERENTIEL_PAC.ichn.animale.ugbMinimum && !estRenseigne(profilExploitation.autresAnimaux.ugbDeclareesHorsBovins)) manquants.push("UGB hors bovins si l'exploitation possède d'autres animaux");
    if (estZoneHorsMontagneICHN() && i.siegeEnZoneDefavorisee === null) manquants.push("Localisation du siège d'exploitation");
    if (manquants.length) return creerResultatAide(id, STATUT_AIDE.INFORMATIONS_MANQUANTES, [], manquants);
    if (nombreOuZero(i.surfacesFourrageresEligibles) < REFERENTIEL_PAC.ichn.animale.surfaceFourragereMinimum) return nonRetenue(id, "La surface fourragère en zone défavorisée est inférieure à 3 ha.");
    if (ugb < REFERENTIEL_PAC.ichn.animale.ugbMinimum) return nonRetenue(id, "Le seuil de 5 UGB n'est pas atteint.");
    if (estZoneHorsMontagneICHN() && i.siegeEnZoneDefavorisee === false) return nonRetenue(id, "Pour l'ICHN animale hors montagne, le siège doit être situé en zone défavorisée.");
    if (estZoneHorsMontagneICHN() && nombreOuZero(i.pourcentageSAUZoneDefavorisee) < 80) return nonRetenue(id, "Moins de 80 % de la SAU est en zone défavorisée : les surfaces hors montagne ne sont alors pas indemnisées selon le guide.");
    const verifications = ["Le montant variable et les plages de chargement sont fixés localement."];
    if (estZoneMontagneICHN() && nombreOuZero(i.pourcentageSAUZoneDefavorisee) < 80) verifications.push("En montagne, un paiement fortement réduit peut subsister lorsque moins de 80 % de la SAU est en zone défavorisée.");
    if (nombreOuZero(i.partRevenuAgricolePourcent) < 50) verifications.push("La part de revenu agricole est inférieure à 50 % : des plafonnements liés au revenu non agricole doivent être vérifiés.");
    return creerResultatAide(id, STATUT_AIDE.A_EXAMINER, ["Au moins 3 ha de surface fourragère en zone défavorisée.", `Au moins 5 UGB disponibles (${Math.round(ugb * 10) / 10} calculées / renseignées).`], [], verifications, { ugb });
}

const TYPE_CALCUL = {
    DONNEE_EXPLOITATION: "donnee_exploitation",
    REFERENCE_2024: "reference_2024",
    REFERENCE_2025: "reference_2025",
    REGLE_GUIDE: "regle_guide",
    PARTIEL: "partiel",
    NON_CALCULABLE: "non_calculable"
};

function arrondirMontant(montant) {
    if (!Number.isFinite(Number(montant))) return null;
    return Math.round(Number(montant) * 100) / 100;
}

function formaterMontant(montant) {
    if (montant === null || montant === undefined || !Number.isFinite(Number(montant))) return "—";
    return new Intl.NumberFormat("fr-FR", { style: "currency", currency: "EUR" }).format(Number(montant));
}

function creerCalcul({ id, nom = null, montant = null, type = TYPE_CALCUL.NON_CALCULABLE, anneeReference = null, formule = null, donnees = {}, avertissements = [], details = {} }) {
    const aide = obtenirAide(id);
    const valeur = montant === null ? null : arrondirMontant(montant);
    return {
        id,
        nom: nom || (aide ? aide.nom : id),
        montant: valeur,
        montantFormate: formaterMontant(valeur),
        type,
        anneeReference,
        formule,
        donnees,
        avertissements,
        details
    };
}

function orientationValide(orientation) {
    return orientation && orientation.statut === STATUT_AIDE.A_EXAMINER;
}

    return creerCalcul({
        id: orientation.id,
        nom: orientation.nom,
        type: TYPE_CALCUL.NON_CALCULABLE,
        avertissements: orientation.informationsManquantes.length ? orientation.informationsManquantes : orientation.verifications
    });
}

function calculerAideBase() {
    const orientation = analyserAideBase();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);
    const surface = nombreOuZero(profilExploitation.surfaces.surfaceAdmissible);
    const dpbActifs = nombreOuZero(profilExploitation.dpb.nombreDPBActives);
    const nombrePrime = Math.min(surface, dpbActifs);
    const total = profilExploitation.dpb.valeurTotaleDPB;
    const moyenne = profilExploitation.dpb.valeurMoyenneDPB;

    if (estRenseigne(total) && Number(total) >= 0) {
        return creerCalcul({ id: "aide_base", montant: Number(total), type: TYPE_CALCUL.DONNEE_EXPLOITATION, formule: "Valeur totale des DPB renseignée", donnees: { surface, dpbActifs } });
    }
    if (estRenseigne(moyenne) && Number(moyenne) >= 0) {
        return creerCalcul({ id: "aide_base", montant: nombrePrime * Number(moyenne), type: TYPE_CALCUL.DONNEE_EXPLOITATION, formule: "min(DPB activés, surface admissible) × valeur moyenne réelle des DPB", donnees: { nombrePrime, valeurMoyenne: Number(moyenne) } });
    }
    return creerCalcul({ id: "aide_base", avertissements: ["Renseignez la valeur moyenne ou la valeur totale réelle de vos DPB. La moyenne nationale n'est pas utilisée comme valeur de votre exploitation."] });

function calculerEcoregime() {
    const orientation = analyserEcoregime();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);
    const details = orientation.details;
    const niveau = details.niveau;
    const ref = REFERENTIEL_PAC.montantsReference.ecoregime2024;
    let tarif = null;
    if (niveau === "bio") tarif = ref.niveauBio;
    else if (niveau === "superieur") tarif = ref.niveauSuperieur;
    else tarif = ref.niveauBase;

    if (niveau === "base" && tarif === null) {
        return creerCalcul({
            id: "ecoregime",
            type: TYPE_CALCUL.NON_CALCULABLE,
            details,
            avertissements: ["Le guide comporte deux valeurs 2024 différentes pour le niveau de base (48,23 et 48,35 €/ha). Aucune n'est choisie automatiquement."]
        });
    }

    const surface = nombreOuZero(profilExploitation.surfaces.surfaceAdmissible);
    let montant = surface * tarif;
    const avertissements = ["Montant calculé avec les références 2024 du guide, pas avec un taux courant automatique."];

    const voie = details.voie;
    if (details.bonusHaies && details.bonusHaies.possible) {
        if (["pratiques", "certification"].includes(voie)) {
            montant += surface * ref.bonusHaies;
            avertissements.push("Bonus haies ajouté avec la référence 2024.");
        } else {
            avertissements.push("Le bonus haies paraît possible via une voie pratiques/certification, mais il n'est pas ajouté car la meilleure voie détectée est la biodiversité.");
        }
    }

    return creerCalcul({
        id: "ecoregime",
        montant,
        type: TYPE_CALCUL.REFERENCE_2024,
        anneeReference: 2024,
        formule: "Surface admissible × référence du niveau détecté, avec éventuel bonus haies",
        donnees: { surface, niveau, voie, tarif, pointsDiversification: details.scoreDiversification?.points },
        avertissements,
        details
    });
}

function calculerICHNAnimale() {
    const orientation = analyserICHNAnimale();
    if (!orientationValide(orientation)) return calculNonEligible(orientation);
    const i = profilExploitation.ichn;
    const p = REFERENTIEL_PAC.ichn.animale;
    const surface = Math.min(nombreOuZero(i.surfacesFourrageresEligibles), p.surfaceMaximum);
    const partFixe = surface * p.montantBaseParHa;
    if (!estRenseigne(i.montantVariableSousZone) || !estRenseigne(i.coefficientModulationChargement)) {
        return creerCalcul({ id: "ichn_animale", type: TYPE_CALCUL.NON_CALCULABLE, formule: "Part fixe + part variable locale, modulées selon le chargement", donnees: { surface, partFixeTheorique: arrondirMontant(partFixe) }, avertissements: [`Part fixe théorique avant modulation : ${formaterMontant(partFixe)}.`, "Le montant variable de sous-zone et le coefficient de chargement sont fixés localement."] });
    }
    const tranche1 = Math.min(surface, p.trancheVariablePleine);
    const tranche2 = Math.max(0, Math.min(surface, p.trancheVariableReduiteFin) - p.trancheVariablePleine);
    const variable = tranche1 * Number(i.montantVariableSousZone) + tranche2 * Number(i.montantVariableSousZone) * p.coefficientDeuxiemeTranche;
    const montant = (partFixe + variable) * Number(i.coefficientModulationChargement);
    return creerCalcul({ id: "ichn_animale", montant, type: TYPE_CALCUL.PARTIEL, formule: "(part fixe + part variable locale) × coefficient de chargement", donnees: { surface, partFixe, partVariable: variable, coefficient: Number(i.coefficientModulationChargement) }, avertissements: ["À confirmer avec l'arrêté préfectoral et les règles locales de chargement / revenu."] });
}

    return creerCalcul({ id: "assurance_mrc", montant: Number(profilExploitation.assurance.primeAnnuelle) * taux, type: TYPE_CALCUL.REFERENCE_2025, anneeReference: 2025, formule: "Prime × taux de prise en charge de référence 2025", donnees: { prime: Number(profilExploitation.assurance.primeAnnuelle), taux }, avertissements: ["Référence 2025 uniquement ; elle n'est pas présentée comme taux courant."] });
}

    return creerCalcul({ id, type: TYPE_CALCUL.NON_CALCULABLE, avertissements: [message] });
}

    return [
        calculerAideBase(), calculerEcoregime(), calculerAideRedistributive(), calculerACJA(),
        calculerLegumineusesGraines(), calculerLegumineusesFourrageres(), calculerBleDur(), calculerPommeTerreFeculiere(), calculerRiz(), calculerHoublon(), calculerSemencesGraminees(), calculerChanvre(), calculerMaraichage(), calculerFruitsTransformes(),
        calculerBovins(), calculerVeauxQualite(), calculerOvins(), calculerCaprins(), calculerBovinsCorse(), calculerPetitsRuminantsCorse(),
        calculerICHNAnimale(), calculerICHNVegetale(), calculerCAB(), calculerMAEC(), calculerAssuranceMRC(), calculerPredation(), calculerProgrammesSectoriels(), calculerAidesRegionales()
    ];
}

function calculerSyntheseFinanciere() {
    const totaux = { donneeExploitation: 0, reference2024: 0, reference2025: 0, regleGuide: 0, partiel: 0 };
    calculerToutesLesAides().forEach(c => {
        if (c.montant === null) return;
        if (c.type === TYPE_CALCUL.DONNEE_EXPLOITATION) totaux.donneeExploitation += c.montant;
        else if (c.type === TYPE_CALCUL.REFERENCE_2024) totaux.reference2024 += c.montant;
        else if (c.type === TYPE_CALCUL.REFERENCE_2025) totaux.reference2025 += c.montant;
        else if (c.type === TYPE_CALCUL.REGLE_GUIDE) totaux.regleGuide += c.montant;
        else if (c.type === TYPE_CALCUL.PARTIEL) totaux.partiel += c.montant;
    });
    Object.keys(totaux).forEach(cle => { totaux[cle] = arrondirMontant(totaux[cle]); });
    return totaux;
}

function convertirValeur(valeur, type) {
    if (valeur === "") return null;
    if (type === "number") {
        const n = Number(valeur);
        return Number.isFinite(n) ? n : null;
    }
    if (valeur === "true") return true;
    if (valeur === "false") return false;
    return valeur;
}

function afficherValeur(v) {
    return v === null || v === undefined ? "" : v;
}

function creerChampQuestion(question) {
    if (question.type === "heading") {
        const bloc = document.createElement("div");
        bloc.className = "form-section-heading";
        bloc.innerHTML = `<h3>${question.titre}</h3>${question.texte ? `<p>${question.texte}</p>` : ""}`;
        return bloc;
    }

    const valeur = lireProfil(question.chemin);
    const wrapper = document.createElement("div");
    wrapper.className = "question-item";

    const zoneLibelle = document.createElement("div");
    zoneLibelle.className = "question-copy";
    const label = document.createElement("label");
    label.className = "question-label";
    label.textContent = question.label;
    zoneLibelle.appendChild(label);
    if (question.aide) {
        const aide = document.createElement("p");
        aide.className = "question-help";
        aide.textContent = question.aide;
        zoneLibelle.appendChild(aide);
    }
    wrapper.appendChild(zoneLibelle);

    const controle = document.createElement("div");
    controle.className = "question-control";

    if (question.type === "yesno") {
        const groupe = document.createElement("div");
        groupe.className = "yesno-group";
        [{ texte: "Oui", valeur: true }, { texte: "Non", valeur: false }].forEach(option => {
            const bouton = document.createElement("button");
            bouton.type = "button";
            bouton.className = "answer-button";
            if (valeur === option.valeur) bouton.classList.add("selected");
            bouton.textContent = option.texte;
            bouton.addEventListener("click", () => {
                modifierProfil(question.chemin, option.valeur);
                sauvegarderProfilLocalement();
                afficherEtape();
            });
            groupe.appendChild(bouton);
        });
        controle.appendChild(groupe);
    } else if (question.type === "select") {
        const select = document.createElement("select");
        (question.options || []).forEach(option => {
            const opt = document.createElement("option");
            opt.value = option.valeur;
            opt.textContent = option.texte;
            if (String(valeur ?? "") === String(option.valeur)) opt.selected = true;
            select.appendChild(opt);
        });
        select.addEventListener("change", event => {
            modifierProfil(question.chemin, convertirValeur(event.target.value, "select"));
            sauvegarderProfilLocalement();
            if (question.refreshOnChange) afficherEtape();
            else mettreAJourPanneau();
        });
        controle.appendChild(select);
    } else {
        const ligne = document.createElement("div");
        ligne.className = "input-with-suffix";
        const input = document.createElement("input");
        input.type = question.type;
        input.value = afficherValeur(valeur);
        if (question.placeholder) input.placeholder = question.placeholder;
        if (question.min !== undefined) input.min = question.min;
        if (question.max !== undefined) input.max = question.max;
        if (question.step !== undefined) input.step = question.step;
        input.addEventListener("change", event => {
            modifierProfil(question.chemin, convertirValeur(event.target.value, question.type));
            sauvegarderProfilLocalement();
            if (question.refreshOnChange) afficherEtape();
            else mettreAJourPanneau();
        });
        ligne.appendChild(input);
        if (question.suffixe) {
            const suffixe = document.createElement("span");
            suffixe.className = "input-suffix";
            suffixe.textContent = question.suffixe;
            ligne.appendChild(suffixe);
        }
        controle.appendChild(ligne);
    }

    wrapper.appendChild(controle);
    return wrapper;
}

function texteNiveau(n) {
    return n === "bio" ? "niveau spécifique AB" : n === "superieur" ? "niveau supérieur" : n === "base" ? "niveau de base" : "non atteint";
}

function htmlPanneauActif() {
    const a = calculerAgriculteurActif();
    if (a.statut === true) return `<div class="live-result ok"><strong>Agriculteur actif : critère initialement compatible</strong><span>${a.raisons.join(" ")}</span></div>`;
    if (a.statut === false) return `<div class="live-result no"><strong>Agriculteur actif : critère non rempli avec les réponses actuelles</strong><span>${a.raisons.join(" ")}</span></div>`;
    return `<div class="live-result pending"><strong>Agriculteur actif : à déterminer</strong><span>À compléter : ${a.manquants.join(" · ")}</span></div>`;
}

function htmlPanneauAssolement() {
    const s = calculerScoreDiversification();
    const classe = s.complet ? (s.points >= 5 ? "ok" : s.points >= 4 ? "warn" : "no") : "pending";
    const details = s.details.map(d => `${d.libelle} +${d.points}`).join(" · ") || "Aucun point calculé pour l'instant";
    return `<div class="live-result ${classe}"><strong>Score de diversification calculé : ${s.points} point(s)</strong><span>${s.complet ? details : `Répartition à compléter. Somme saisie : ${Math.round(s.sommeAssolement * 100) / 100} ha pour ${Math.round(s.terresArables * 100) / 100} ha de TA.`}</span></div>`;
}

function htmlPanneauElevage() {
    const u = calculerUGBBovines();
    const autres = nombreOuZero(profilExploitation.autresAnimaux.ugbDeclareesHorsBovins);
    return `<div class="live-result info"><strong>UGB bovines calculées : ${Math.round(u.total * 10) / 10}</strong><span>Niveau supérieur brut : ${Math.round(u.niveauSuperieurBrut * 10) / 10} UGB · niveau base brut : ${Math.round(u.niveauBaseBrut * 10) / 10} UGB${autres > 0 ? ` · autres UGB déclarées : ${autres}` : ""}</span></div>`;
}

function htmlPanneauEcoregime() {
    const d = calculerDiagnosticEcoregime();
    const score = d.scoreDiversification;
    const lignes = [];
    if (d.voies.pratiques.eligible) lignes.push(`Pratiques : ${texteNiveau(d.voies.pratiques.niveau)}`);
    else if (d.voies.pratiques.manquants?.length) lignes.push("Pratiques : informations à compléter");
    else lignes.push("Pratiques : niveau de base non atteint");
    if (d.voies.certification.eligible) lignes.push(`Certification : ${texteNiveau(d.voies.certification.niveau)}`);
    if (d.voies.biodiversite.eligible) lignes.push(`Biodiversité : ${texteNiveau(d.voies.biodiversite.niveau)}`);
    else if (d.voies.biodiversite.iae?.ratioSAU !== null && d.voies.biodiversite.iae?.donneesComplete) lignes.push(`Biodiversité : ${Math.round(d.voies.biodiversite.iae.ratioSAU * 1000) / 10} % IAE/jachères`);
    const meilleur = d.meilleureVoie ? `Meilleure voie détectée : ${d.meilleureVoie.voie} — ${texteNiveau(d.meilleureVoie.niveau)}` : "Aucune voie confirmée pour l'instant";
    return `<div class="live-result ${d.meilleureVoie ? "ok" : "pending"}"><strong>${meilleur}</strong><span>Score diversification : ${score.points} point(s). ${lignes.join(" · ")}${d.bonusHaies.possible ? " · Bonus haies potentiellement accessible." : ""}</span></div>`;
}

function htmlPanneauSynthese() {
    const d = calculerIndicateursPAC();
    const eco = d.ecoregime.meilleureVoie ? `${d.ecoregime.meilleureVoie.voie} / ${texteNiveau(d.ecoregime.meilleureVoie.niveau)}` : "à compléter";
    const actif = d.agriculteurActif.statut === true ? "compatible" : d.agriculteurActif.statut === false ? "non" : "à compléter";
    return `<div class="live-result info"><strong>Pré-diagnostic automatique</strong><span>Agriculteur actif : ${actif} · Écorégime : ${eco} · UGB bovines : ${Math.round(d.bovins.total * 10) / 10} · Secteurs détectés : ${d.secteursPresents.length ? d.secteursPresents.join(", ") : "aucun"}</span></div>`;
}

function mettreAJourPanneau() {
    const zone = document.getElementById("live-calculation");
    if (!zone) return;
    const etape = ETAPES_QUESTIONNAIRE[etapeQuestionnaire];
    if (etape.panneau === "actif") zone.innerHTML = htmlPanneauActif();
    else if (etape.panneau === "assolement") zone.innerHTML = htmlPanneauAssolement();
    else if (etape.panneau === "elevage") zone.innerHTML = htmlPanneauElevage();
    else if (etape.panneau === "ecoregime") zone.innerHTML = htmlPanneauEcoregime();
    else if (etape.panneau === "synthese") zone.innerHTML = htmlPanneauSynthese();
    else zone.innerHTML = "";
    zone.style.display = zone.innerHTML ? "block" : "none";
}

function afficherNavigationLaterale() {
    const navigation = document.getElementById("sidebar-steps");
    if (!navigation) return;
    navigation.innerHTML = "";
    ETAPES_QUESTIONNAIRE.forEach((etape, index) => {
        const bouton = document.createElement("button");
        bouton.type = "button";
        bouton.className = "sidebar-step";
        if (!affichageResultats && index === etapeQuestionnaire) bouton.classList.add("active");
        if (index < etapeQuestionnaire || affichageResultats) bouton.classList.add("completed");
        bouton.innerHTML = `<span class="step-indicator">${index < etapeQuestionnaire || affichageResultats ? "✓" : index + 1}</span><span>${etape.titre}</span>`;
        bouton.addEventListener("click", () => {
            affichageResultats = false;
            etapeQuestionnaire = index;
            document.getElementById("results").style.display = "none";
            document.getElementById("questionnaire-card").style.display = "block";
            afficherEtape();
        });
        navigation.appendChild(bouton);
    });

    const resultat = document.createElement("button");
    resultat.type = "button";
    resultat.className = `sidebar-step ${affichageResultats ? "active" : ""}`;
    resultat.innerHTML = `<span class="step-indicator">${affichageResultats ? "✓" : ETAPES_QUESTIONNAIRE.length + 1}</span><span>Mon diagnostic PAC</span>`;
    if (!affichageResultats) {
        resultat.disabled = true;
        resultat.setAttribute("aria-disabled", "true");
    }
    navigation.appendChild(resultat);
}

function afficherEtape() {
    affichageResultats = false;
    const etape = ETAPES_QUESTIONNAIRE[etapeQuestionnaire];
    afficherNavigationLaterale();
    document.getElementById("question-title").textContent = etape.titre;
    document.getElementById("question-subtitle").textContent = etape.sousTitre;

    const zone = document.getElementById("questions-container");
    zone.innerHTML = "";
    etape.questions.forEach(question => {
        if (question.afficherSi && !question.afficherSi()) return;
        zone.appendChild(creerChampQuestion(question));
    });

    mettreAJourPanneau();

    const pourcentage = ((etapeQuestionnaire + 1) / ETAPES_QUESTIONNAIRE.length) * 100;
    document.getElementById("progress-bar").style.width = `${pourcentage}%`;
    document.getElementById("step-counter").textContent = `Rubrique ${etapeQuestionnaire + 1} sur ${ETAPES_QUESTIONNAIRE.length}`;
    document.getElementById("previous-button").style.display = etapeQuestionnaire === 0 ? "none" : "inline-flex";
    document.getElementById("next-button").textContent = etapeQuestionnaire === ETAPES_QUESTIONNAIRE.length - 1 ? "Calculer mon diagnostic" : "Enregistrer et continuer →";
    window.scrollTo({ top: 0, behavior: "smooth" });
}

function etapeSuivante() {
    sauvegarderProfilLocalement();
    if (etapeQuestionnaire < ETAPES_QUESTIONNAIRE.length - 1) {
        etapeQuestionnaire += 1;
        afficherEtape();
    } else {
        afficherResultatsQuestionnaire();
    }
}

function etapePrecedente() {
    if (etapeQuestionnaire > 0) {
        etapeQuestionnaire -= 1;
        afficherEtape();
    }
}

function creerCarteResultat(resultat, calcul) {
    const carte = document.createElement("article");
    carte.className = `result-card ${resultat.statut === STATUT_AIDE.A_EXAMINER ? "success" : "warning"}`;

        ? `<div class="result-amount">${calcul.montantFormate}</div><div class="result-reference">${libelleTypeCalcul(calcul.type)}</div>`
        : `<div class="result-reference">${calcul ? libelleTypeCalcul(calcul.type) : ""}</div>`;

    const avertissements = [...new Set([...(resultat.verifications || []), ...((calcul && calcul.avertissements) || [])])];

const CHATBOT_STORAGE_LANGUAGE = "assistant_pac_chat_language";
const CHATBOT_STORAGE_HISTORY = "assistant_pac_chat_history";

let chatbotLanguage = "fr";
let chatbotHistory = [];
let chatbotBusy = false;

function getChatbotLanguage() {
    return chatbotLanguage;
}

function normaliserTexteChatbot(texte) {
    return String(texte || "")
        .toLowerCase()
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .replace(/[^a-z0-9áéíóúñüàâäçèêëîïôöùûüœæ\s-]/gi, " ")
        .replace(/\s+/g, " ")
        .trim();
}

function chargerEtatChatbot() {
    const savedLanguage = localStorage.getItem(CHATBOT_STORAGE_LANGUAGE);
    if (CHATBOT_LANGUAGES.includes(savedLanguage)) chatbotLanguage = savedLanguage;

    try {
        const history = JSON.parse(localStorage.getItem(CHATBOT_STORAGE_HISTORY) || "[]");
        if (Array.isArray(history)) chatbotHistory = history.slice(-40);
    } catch (error) {
        chatbotHistory = [];
    }
}

function sauvegarderHistoriqueChatbot() {
    localStorage.setItem(CHATBOT_STORAGE_HISTORY, JSON.stringify(chatbotHistory.slice(-40)));
}

function setChatbotLanguage(lang) {
    if (!CHATBOT_LANGUAGES.includes(lang)) return;
    chatbotLanguage = lang;
    localStorage.setItem(CHATBOT_STORAGE_LANGUAGE, lang);
    mettreAJourTextesChatbot();
    afficherHistoriqueChatbot();
}

function formaterVoieEcoregimeChatbot(voie) {
    if (!voie) return null;
    const map = {
        pratiques: chatbotT("bestWayPractices"),
        certification: chatbotT("bestWayCertification"),
        biodiversite: chatbotT("bestWayBiodiversity")
    };
    return map[voie] || voie;
}

function formaterNiveauEcoregimeChatbot(niveau) {
    const map = {
        base: chatbotT("levelBase"),
        superieur: chatbotT("levelHigher"),
        bio: chatbotT("levelBio")
    };
    return map[niveau] || niveau || "";
}

function obtenirIndicateursChatbot() {

function detecterIntentionChatbot(message) {
    const q = normaliserTexteChatbot(message);
    const has = (...terms) => terms.some(term => q.includes(normaliserTexteChatbot(term)));

    if (has("aide", "aides", "support", "scheme", "schemes", "ayuda", "ayudas", "diagnostic", "resultat", "resultats", "results", "resultado", "resultados", "eligible", "eligibility", "elegible") &&
        has("mon", "mes", "my", "mi", "mis", "quelles", "which", "cuales", "combien", "how much", "cuanto")) return "diagnostic";

    if (has("ecoregime", "eco regime", "eco-scheme", "ecoscheme", "ecorregimen", "ecorregímen", "ecorregimenes", "points", "diversification", "biodiversite", "biodiversity", "biodiversidad")) return "ecoregime";
    if (has("ugb", "livestock unit", "livestock units", "lu ", "ugm", "unidad ganadera")) return "ugb";
    if (has("agriculteur actif", "active farmer", "agricultor activo")) return "actif";
    if (has("dpb", "droit paiement", "droits paiement", "payment entitlement", "payment entitlements", "derechos de pago", "aide de base", "basic income")) return "dpb";
    if (has("redistributive", "redistributif", "redistributiva")) return "redistributive";
    if (has("jeune agriculteur", "young farmer", "joven agricultor", "acja")) return "young";
    if (has("ichn", "anc", "handicap naturel", "natural constraint", "zona desfavorecida")) return "ichn";
    if (has("cab", "conversion bio", "conversion biologique", "organic conversion", "conversion ecologica", "conversion ecológica")) return "bio";
    if (has("maec", "agri-environment", "agri environment", "agroambiental", "climate measure")) return "maec";
    if (has("bovin", "bovine", "cattle", "cow", "sheep", "ovin", "goat", "caprin", "ganado", "oveja", "cabra", "veau", "calf", "ternero")) return "animal";
    if (has("legumineuse", "legume", "protein crop", "crop support", "aide vegetale", "vegetal", "trigo duro", "ble dur", "riz", "rice", "arroz", "chanvre", "hemp", "canamo", "cáñamo", "maraichage", "horticulture")) return "plant";
    if (has("source", "referentiel", "reference", "guide", "fuente")) return "source";
    if (has("aide", "help", "ayuda", "que peux tu", "what can you", "que puedes")) return "help";
    return "unknown";
}

function reponseDiagnosticChatbot() {
    const orientations = obtenirOrientationsChatbot();
    const calculs = obtenirCalculsChatbot();
    const statutExaminer = typeof STATUT_AIDE !== "undefined" ? STATUT_AIDE.A_EXAMINER : "A_EXAMINER";
    const aExaminer = orientations.filter(item => item.statut === statutExaminer);

    let texte = aExaminer.length
        ? chatbotT("diagnosticIntro", {
            count: aExaminer.length,
            list: aExaminer.slice(0, 8).map(item => item.nom).join(", ")
        })
        : chatbotT("diagnosticNone");

    const calculables = calculs
        .filter(item => typeof item.montant === "number" && Number.isFinite(item.montant) && item.montant > 0)
        .slice(0, 8);

    if (calculables.length) {
        const liste = calculables.map(item => {
            const montant = item.montantFormate || (typeof formaterMontant === "function" ? formaterMontant(item.montant) : `${Math.round(item.montant * 100) / 100} €`);
            return `${item.nom || item.id}: ${montant}`;
        }).join(" · ");
        texte += `\n\n${chatbotT("amountIntro", { list: liste })}`;
    } else {
        texte += `\n\n${chatbotT("amountNone")}`;
    }

    return texte;
}

function genererReponseLocaleChatbot(message) {
    const intention = detecterIntentionChatbot(message);
    const indicateurs = obtenirIndicateursChatbot();

    if (intention === "ecoregime") {
        let texte = chatbotT("ecoGeneric");
        const eco = indicateurs?.ecoregime;
        if (eco?.meilleureVoie) {
            const result = `${formaterVoieEcoregimeChatbot(eco.meilleureVoie.voie)} — ${formaterNiveauEcoregimeChatbot(eco.meilleureVoie.niveau)}`;
            texte += `\n\n${chatbotT("ecoCurrent", { result })}`;
            if (eco.scoreDiversification && typeof eco.scoreDiversification.points === "number") {
                const label = chatbotLanguage === "en" ? "Diversification score" : chatbotLanguage === "es" ? "Puntuación de diversificación" : "Score de diversification";
                texte += ` ${label}: ${eco.scoreDiversification.points}.`;
            }
        } else {
            texte += `\n\n${chatbotT("ecoNoResult")}`;
        }
        return texte;
    }

    if (intention === "ugb") {
        const value = indicateurs?.bovins?.total;
        return typeof value === "number"
            ? chatbotT("ugbCurrent", { value: Math.round(value * 10) / 10 })
            : chatbotT("noData");
    }

    if (intention === "actif") {
        const statut = indicateurs?.agriculteurActif?.statut;
        if (statut === true) return chatbotT("activeTrue");
        if (statut === false) return chatbotT("activeFalse");
        return chatbotT("activeUnknown");
    }

    if (intention === "diagnostic") return reponseDiagnosticChatbot();
    if (["dpb", "redistributive", "young", "ichn", "bio", "maec", "animal", "plant", "help", "source"].includes(intention)) return chatbotT(intention);
    return `${chatbotT("unknown")}\n\n${chatbotT("help")}`;
}

async function appelerBackendChatbot(message) {
    const config = window.CHATBOT_CONFIG || {};
    if (!config.apiEndpoint) return null;

    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), config.timeoutMs || 15000);

    try {
        const payload = {
            message,
            language: chatbotLanguage,
            history: chatbotHistory.slice(-12),
            profile: typeof profilExploitation !== "undefined" ? profilExploitation : null,
            indicators: obtenirIndicateursChatbot()
        };

        const response = await fetch(config.apiEndpoint, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(payload),
            signal: controller.signal
        });

        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const data = await response.json();
        return data.answer || data.message || data.response || null;
    } finally {
        clearTimeout(timer);
    }
}

function afficherHistoriqueChatbot() {
    const zone = document.getElementById("chatbot-messages");
    if (!zone) return;
    zone.innerHTML = "";

    if (!chatbotHistory.length) {
        zone.appendChild(creerMessageChatbot("assistant", chatbotT("welcome")));
    } else {
        chatbotHistory.forEach(item => zone.appendChild(creerMessageChatbot(item.role, item.text)));
    }
    zone.scrollTop = zone.scrollHeight;
}

function ajouterMessageHistorique(role, text) {
    chatbotHistory.push({ role, text, language: chatbotLanguage, at: new Date().toISOString() });
    chatbotHistory = chatbotHistory.slice(-40);
    sauvegarderHistoriqueChatbot();
    afficherHistoriqueChatbot();
}

async function envoyerMessageChatbot() {
    if (chatbotBusy) return;
    const input = document.getElementById("chatbot-input");
    const message = input?.value.trim();
    if (!message) return;

    input.value = "";
    ajouterMessageHistorique("user", message);
    chatbotBusy = true;
    afficherEtatSaisieChatbot(true);

    let reponse = null;
    try {
        reponse = await appelerBackendChatbot(message);
    } catch (error) {
        console.warn("Chatbot backend indisponible", error);
        ajouterMessageHistorique("assistant", chatbotT("backendError"));
    }

    if (!reponse) reponse = genererReponseLocaleChatbot(message);
    ajouterMessageHistorique("assistant", reponse);

    afficherEtatSaisieChatbot(false);
    chatbotBusy = false;
}

function effacerConversationChatbot() {
    chatbotHistory = [];
    localStorage.removeItem(CHATBOT_STORAGE_HISTORY);
    afficherHistoriqueChatbot();
}

    chargerEtatChatbot();
    mettreAJourTextesChatbot();
    afficherHistoriqueChatbot();

    document.getElementById("chatbot-launcher")?.addEventListener("click", ouvrirChatbot);
    document.getElementById("chatbot-close")?.addEventListener("click", fermerChatbot);
    document.getElementById("chatbot-send")?.addEventListener("click", envoyerMessageChatbot);
    document.getElementById("chatbot-clear")?.addEventListener("click", effacerConversationChatbot);

    document.querySelectorAll("[data-chat-lang]").forEach(button => {
        button.addEventListener("click", () => setChatbotLanguage(button.dataset.chatLang));
    });

    document.getElementById("chatbot-input")?.addEventListener("keydown", event => {
        if (event.key === "Enter" && !event.shiftKey) {
            event.preventDefault();
            envoyerMessageChatbot();
        }
    });
}
